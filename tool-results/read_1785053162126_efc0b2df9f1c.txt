    92→    92→function sanitizePhone(raw: string | null | undefined): string {
    93→    93→  if (!raw) return "";
    94→    94→  return raw.replace(/[^\d+]/g, "").replace(/(^\+?)(\+)+/, "$1");
    95→    95→}
    96→    96→
    97→    97→function whatsappLink(raw: string | null | undefined): string {
    98→    98→  const digits = sanitizePhone(raw).replace(/^\+/, "").replace(/^0+/, "");
    99→    99→  return `https://wa.me/${digits}`;
   100→   100→}
   101→   101→
   102→   102→function telLink(raw: string | null | undefined): string {
   103→   103→  return `tel:${sanitizePhone(raw)}`;
   104→   104→}

   323→   323→
   324→   324→function CustomerFormDialog({
   325→   325→  open, onOpenChange, customer,
   326→   326→}: {
   327→   327→  open: boolean;
   328→   328→  onOpenChange: (o: boolean) => void;
   329→   329→  customer: Customer | null;
   330→   330→}) {
   331→   331→  // Keyed remount strategy: the inner form holds its own state initialized
   332→   332→  // from `customer`. When `customer` (or open state) changes, React remounts
   333→   333→  // the inner form, resetting state cleanly without setState-in-effect.
   334→   334→  return (
   335→   335→    <Dialog open={open} onOpenChange={onOpenChange}>
   336→   336→      <DialogContent className="max-w-2xl gap-0 p-0">
   337→   337→        <CustomerFormInner
   338→   338→          key={open ? (customer?.id ?? "new") : "closed"}
   339→   339→          customer={customer}
   340→   340→          onOpenChange={onOpenChange}
   341→   341→        />
   342→   342→      </DialogContent>
   343→   343→    </Dialog>
   344→   344→  );
   345→   345→}
   346→   346→
   347→   347→function CustomerFormInner({
   348→   348→  customer, onOpenChange,
   349→   349→}: {
   350→   350→  customer: Customer | null;
   351→   351→  onOpenChange: (o: boolean) => void;
   352→   352→}) {
   353→   353→  const qc = useQueryClient();
   354→   354→  const [form, setForm] = useState<CustomerFormValues>(() =>
   355→   355→    customer
   356→   356→      ? {
   357→   357→          id: customer.id,
   358→   358→          name: customer.name,
   359→   359→          phone: customer.phone ?? "",
   360→   360→          whatsapp: customer.whatsapp ?? "",
   361→   361→          email: customer.email ?? "",
   362→   362→          address: customer.address ?? "",
   363→   363→          company: customer.company ?? "",
   364→   364→          notes: customer.notes ?? "",
   365→   365→        }
   366→   366→      : emptyForm
   367→   367→  );
   368→   368→
   369→   369→  const save = useMutation({
   370→   370→    mutationFn: async () => {
   371→   371→      if (form.id) return api.put(`/customers/${form.id}`, form);
   372→   372→      return api.post("/customers", form);
   373→   373→    },
   374→   374→    onSuccess: () => {
   375→   375→      qc.invalidateQueries({ queryKey: ["customers"] });
   376→   376→      toast.success(form.id ? "Customer updated" : "Customer created");
   377→   377→      onOpenChange(false);
   378→   378→    },
   379→   379→    onError: (e: Error) => toast.error(e.message),
   380→   380→  });
   381→   381→
   382→   382→  const set = (k: keyof CustomerFormValues, v: string) =>
   383→   383→    setForm((f) => ({ ...f, [k]: v }));
   384→   384→
   385→   385→  return (
   386→   386→    <>
   387→   387→      <DialogHeader className="border-b px-6 py-4">
   388→   388→        <DialogTitle>{form.id ? "Edit Customer" : "New Customer"}</DialogTitle>
   389→   389→      </DialogHeader>
   390→   390→        <ScrollArea className="max-h-[70vh]">
   391→   391→          <div className="space-y-4 px-6 py-5">
   392→   392→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   393→   393→              <Field label="Customer Name *">
   394→   394→                <Input value={form.name} onChange={(e) => set("name", e.target.value)} placeholder="e.g. John Doe" />
   395→   395→              </Field>
   396→   396→              <Field label="Company">
   397→   397→                <Input value={form.company} onChange={(e) => set("company", e.target.value)} placeholder="Shop / business name" />
   398→   398→              </Field>
   399→   399→              <Field label="Phone">
   400→   400→                <Input value={form.phone} onChange={(e) => set("phone", e.target.value)} placeholder="+92 300 1234567" />
   401→   401→              </Field>
   402→   402→              <Field label="WhatsApp">
   403→   403→                <Input value={form.whatsapp} onChange={(e) => set("whatsapp", e.target.value)} placeholder="+92 300 1234567" />
   404→   404→              </Field>
   405→   405→              <Field label="Email" className="sm:col-span-2">
   406→   406→                <Input type="email" value={form.email} onChange={(e) => set("email", e.target.value)} placeholder="customer@email.com" />
   407→   407→              </Field>
   408→   408→              <Field label="Address" className="sm:col-span-2">
   409→   409→                <Input value={form.address} onChange={(e) => set("address", e.target.value)} placeholder="Street, City, Country" />
   410→   410→              </Field>
   411→   411→              <Field label="Notes" className="sm:col-span-2">
   412→   412→                <Textarea rows={3} value={form.notes} onChange={(e) => set("notes", e.target.value)} placeholder="Preferences, payment habits, special instructions…" />
   413→   413→              </Field>
   414→   414→            </div>
   415→   415→          </div>
   416→   416→        </ScrollArea>
   417→   417→        <DialogFooter className="border-t px-6 py-4">
   418→   418→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   419→   419→          <Button onClick={() => save.mutate()} disabled={save.isPending || !form.name.trim()}>
   420→   420→            {save.isPending ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   421→   421→            {form.id ? "Save Changes" : "Create Customer"}
   422→   422→          </Button>
   423→   423→        </DialogFooter>
   424→   424→    </>
   425→   425→  );
   426→   426→}
