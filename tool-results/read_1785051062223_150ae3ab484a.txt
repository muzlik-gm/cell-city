    91→// ─── Helpers ─────────────────────────────────────────────────────────────
    92→function sanitizePhone(raw: string | null | undefined): string {
    93→  if (!raw) return "";
    94→  return raw.replace(/[^\d+]/g, "").replace(/(^\+?)(\+)+/, "$1");
    95→}
    96→
    97→function whatsappLink(raw: string | null | undefined): string {
    98→  const digits = sanitizePhone(raw).replace(/^\+/, "").replace(/^0+/, "");
    99→  return `https://wa.me/${digits}`;
   100→}
   101→
   102→function telLink(raw: string | null | undefined): string {
   103→  return `tel:${sanitizePhone(raw)}`;
   104→}

   308→// ─── Customer Form Dialog ────────────────────────────────────────────────
   309→interface CustomerFormValues {
   310→  id?: string;
   311→  name: string;
   312→  phone: string;
   313→  whatsapp: string;
   314→  email: string;
   315→  address: string;
   316→  company: string;
   317→  notes: string;
   318→}
   319→
   320→const emptyForm: CustomerFormValues = {
   321→  name: "", phone: "", whatsapp: "", email: "", address: "", company: "", notes: "",
   322→};
   323→
   324→function CustomerFormDialog({
   325→  open, onOpenChange, customer,
   326→}: {
   327→  open: boolean;
   328→  onOpenChange: (o: boolean) => void;
   329→  customer: Customer | null;
   330→}) {
   331→  // Keyed remount strategy: the inner form holds its own state initialized
   332→  // from `customer`. When `customer` (or open state) changes, React remounts
   333→  // the inner form, resetting state cleanly without setState-in-effect.
   334→  return (
   335→    <Dialog open={open} onOpenChange={onOpenChange}>
   336→      <DialogContent className="max-w-2xl gap-0 p-0">
   337→        <CustomerFormInner
   338→          key={open ? (customer?.id ?? "new") : "closed"}
   339→          customer={customer}
   340→          onOpenChange={onOpenChange}
   341→        />
   342→      </DialogContent>
   343→    </Dialog>
   344→  );
   345→}
   346→
   347→function CustomerFormInner({
   348→  customer, onOpenChange,
   349→}: {
   350→  customer: Customer | null;
   351→  onOpenChange: (o: boolean) => void;
   352→}) {
   353→  const qc = useQueryClient();
   354→  const [form, setForm] = useState<CustomerFormValues>(() =>
   355→    customer
   356→      ? {
   357→          id: customer.id,
   358→          name: customer.name,
   359→          phone: customer.phone ?? "",
   360→          whatsapp: customer.whatsapp ?? "",
   361→          email: customer.email ?? "",
   362→          address: customer.address ?? "",
   363→          company: customer.company ?? "",
   364→          notes: customer.notes ?? "",
   365→        }
   366→      : emptyForm
   367→  );
   368→
   369→  const save = useMutation({
   370→    mutationFn: async () => {
   371→      if (form.id) return api.put(`/customers/${form.id}`, form);
   372→      return api.post("/customers", form);
   373→    },
   374→    onSuccess: () => {
   375→      qc.invalidateQueries({ queryKey: ["customers"] });
   376→      toast.success(form.id ? "Customer updated" : "Customer created");
   377→      onOpenChange(false);
   378→    },
   379→    onError: (e: Error) => toast.error(e.message),
   380→  });
   381→
   382→  const set = (k: keyof CustomerFormValues, v: string) =>
   383→    setForm((f) => ({ ...f, [k]: v }));
   384→
   385→  return (
   386→    <>
   387→      <DialogHeader className="border-b px-6 py-4">
   388→        <DialogTitle>{form.id ? "Edit Customer" : "New Customer"}</DialogTitle>
   389→      </DialogHeader>
   390→        <ScrollArea className="max-h-[70vh]">
   391→          <div className="space-y-4 px-6 py-5">
   392→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   393→              <Field label="Customer Name *">
   394→                <Input value={form.name} onChange={(e) => set("name", e.target.value)} placeholder="e.g. John Doe" />
   395→              </Field>
   396→              <Field label="Company">
   397→                <Input value={form.company} onChange={(e) => set("company", e.target.value)} placeholder="Shop / business name" />
   398→              </Field>
   399→              <Field label="Phone">
   400→                <Input value={form.phone} onChange={(e) => set("phone", e.target.value)} placeholder="+92 300 1234567" />
   401→              </Field>
   402→              <Field label="WhatsApp">
   403→                <Input value={form.whatsapp} onChange={(e) => set("whatsapp", e.target.value)} placeholder="+92 300 1234567" />
   404→              </Field>
   405→              <Field label="Email" className="sm:col-span-2">
   406→                <Input type="email" value={form.email} onChange={(e) => set("email", e.target.value)} placeholder="customer@email.com" />
   407→              </Field>
   408→              <Field label="Address" className="sm:col-span-2">
   409→                <Input value={form.address} onChange={(e) => set("address", e.target.value)} placeholder="Street, City, Country" />
   410→              </Field>
   411→              <Field label="Notes" className="sm:col-span-2">
   412→                <Textarea rows={3} value={form.notes} onChange={(e) => set("notes", e.target.value)} placeholder="Preferences, payment habits, special instructions…" />
   413→              </Field>
   414→            </div>
   415→          </div>
   416→        </ScrollArea>
   417→        <DialogFooter className="border-t px-6 py-4">
   418→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   419→          <Button onClick={() => save.mutate()} disabled={save.isPending || !form.name.trim()}>
   420→            {save.isPending ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   421→            {form.id ? "Save Changes" : "Create Customer"}
   422→          </Button>
   423→        </DialogFooter>
   424→    </>
   425→  );
   426→}

   520→                      <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-teal-500/10 text-teal-600"><Mail className="h-4 w-4" /></div>
   521→                      <div className="min-w-0">
   522→                        <p className="text-[11px] text-muted-foreground">Email</p>
   523→                        <p className="truncate text-sm font-medium">{customer.email}</p>
   524→                      </div>
   525→                    </a>
   526→                  )}
   527→                  {customer.address && (
   528→                    <div className="flex items-center gap-3 rounded-lg border p-3">
   529→                      <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-500/10 text-amber-600"><MapPin className="h-4 w-4" /></div>
   530→                      <div className="min-w-0">
   531→                        <p className="text-[11px] text-muted-foreground">Address</p>
   532→                        <p className="truncate text-sm font-medium">{customer.address}</p>
   533→                      </div>
   534→                    </div>
   535→                  )}
   536→                  {customer.company && (
   537→                    <div className="flex items-center gap-3 rounded-lg border p-3">
   538→                      <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-purple-500/10 text-purple-600"><Building2 className="h-4 w-4" /></div>
   539→                      <div className="min-w-0">
   540→                        <p className="text-[11px] text-muted-foreground">Company</p>
   541→                        <p className="truncate text-sm font-medium">{customer.company}</p>
   542→                      </div>
   543→                    </div>
   544→                  )}
   545→                </div>
   546→
   547→                {/* Notes */}
   548→                {customer.notes && (
   549→                  <div className="mt-4 rounded-xl border border-amber-500/20 bg-amber-500/5 p-4">
   550→                    <p className="mb-1 text-xs font-semibold text-amber-600">Notes</p>
   551→                    <p className="text-sm">{customer.notes}</p>
   552→                  </div>
   553→                )}
   554→
   555→                {/* Tabs */}
   556→                <Tabs defaultValue="sales" className="mt-5">
   557→                  <TabsList className="mb-4">
   558→                    <TabsTrigger value="sales">Purchase History</TabsTrigger>
   559→                    <TabsTrigger value="repairs">Repair History</TabsTrigger>
   560→                  </TabsList>
   561→
   562→                  <TabsContent value="sales">
   563→                    {!customer.sales?.length ? (
   564→                      <EmptyState icon={Receipt} title="No purchases yet" description="Sales invoices for this customer will appear here." />
   565→                    ) : (
   566→                      <div className="space-y-2">
   567→                        {customer.sales.map((s) => (
   568→                          <div key={s.id} className="rounded-lg border p-3">
   569→                            <div className="flex items-center justify-between gap-2">
   570→                              <div className="min-w-0">
   571→                                <div className="flex items-center gap-2">
   572→                                  <p className="text-sm font-semibold">{s.invoiceNo}</p>
   573→                                  <PaymentStatusBadge status={s.paymentStatus} />
   574→                                  <Badge variant="secondary" className="text-[10px]">{s.status}</Badge>
   575→                                  <Badge variant="outline" className="text-[10px]">{s.paymentMethod}</Badge>
   576→                                </div>
   577→                                <p className="text-[11px] text-muted-foreground">{formatDateTime(s.createdAt)} · {s.items?.length ?? 0} items</p>
   578→                              </div>
   579→                              <div className="text-right">
   580→                                <p className="text-sm font-semibold">{formatCurrency(s.total)}</p>
   581→                                <p className="text-[11px] text-muted-foreground">paid {formatCurrency(s.paid)}</p>
   582→                              </div>
   583→                            </div>
   584→                            {s.items?.length > 0 && (
   585→                              <div className="mt-2 space-y-1 border-t pt-2">
   586→                                {s.items.map((it) => (
   587→                                  <div key={it.id} className="flex items-center justify-between text-xs text-muted-foreground">
   588→                                    <span className="truncate">{it.name} <span className="text-[10px]">×{it.qty}</span></span>
   589→                                    <span className="font-medium text-foreground">{formatCurrency(it.total)}</span>
   590→                                  </div>
   591→                                ))}
   592→                              </div>
   593→                            )}
   594→                          </div>
   595→                        ))}
   596→                      </div>
   597→                    )}
   598→                  </TabsContent>
   599→
   600→                  <TabsContent value="repairs">
   601→                    {!customer.repairJobs?.length ? (
   602→                      <EmptyState icon={Wrench} title="No repair jobs yet" description="Repair tickets for this customer will appear here." />
   603→                    ) : (
   604→                      <div className="space-y-2">
   605→                        {customer.repairJobs.map((r) => (
   606→                          <div key={r.id} className="rounded-lg border p-3">
   607→                            <div className="flex items-start justify-between gap-2">
   608→                              <div className="min-w-0">
   609→                                <div className="flex items-center gap-2">
   610→                                  <p className="text-sm font-semibold">{r.ticketNo}</p>
   611→                                  <RepairStatusBadge status={r.status} />
   612→                                  <PaymentStatusBadge status={r.paymentStatus} />
   613→                                </div>
   614→                                <p className="mt-0.5 truncate text-xs font-medium">{r.problem}</p>
   615→                                <p className="text-[11px] text-muted-foreground">
   616→                                  {formatDate(r.createdAt)}
   617→                                  {r.model?.name ? ` · ${r.model.brand?.name ?? ""} ${r.model.name}` : ""}
   618→                                </p>
   619→                              </div>
   620→                              <div className="text-right">
   621→                                <p className="text-sm font-semibold">{formatCurrency(r.total)}</p>
   622→                                <p className="text-[11px] text-muted-foreground">paid {formatCurrency(r.paid)}</p>
   623→                              </div>
   624→                            </div>
   625→                          </div>
   626→                        ))}
   627→                      </div>
   628→                    )}
   629→                  </TabsContent>
   630→                </Tabs>
   631→              </div>
   632→            </ScrollArea>
   633→          </>
   634→        )}
   635→      </SheetContent>
   636→    </Sheet>
   637→  );
   638→}
