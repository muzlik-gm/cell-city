   132→  const [query, setQuery] = useState("");
   133→  const [activePart, setActivePart] = useState<PartKind | "ANY">("ANY");
   134→  const [manageOpen, setManageOpen] = useState(false);
   135→
   136→  // Debounce search input → query
   137→  useEffect(() => {
   138→    const t = setTimeout(() => {
   139→      setQuery(input.trim());
   140→      setActivePart("ANY");
   141→    }, 300);
   142→    return () => clearTimeout(t);
   143→  }, [input]);
   144→
   145→  const result = useQuery<CompatResult>({
   146→    queryKey: ["compat", query],
   147→    queryFn: () => api.get<CompatResult>(`/compatibility?q=${encodeURIComponent(query)}`),

   149→    staleTime: 30_000,
   150→  });

   152→  // Group peers by partType
   153→  const peersByPart = useMemo(() => {
   154→    const map = new Map<string, Peer[]>();
   155→    for (const p of result.data?.peers ?? []) {
   156→      const key = p.partType || "ALL";
   157→      if (!map.has(key)) map.set(key, []);
   158→      map.get(key)!.push(p);
   159→    }
   160→    return map;
   161→  }, [result.data]);
   162→
   163→  // Group products by partType name (the PartType.name like "LCD", "OLED")
   164→  const productsByPart = useMemo(() => {
   165→    const map = new Map<string, Product[]>();
   166→    for (const p of result.data?.products ?? []) {
   167→      const key = p.partType?.name ?? "Other";
   168→      if (!map.has(key)) map.set(key, []);
   169→      map.get(key)!.push(p);
   170→    }
   171→    return map;
   172→  }, [result.data]);
   173→
   174→  // Available part-type chips from current results
   175→  const availablePartTypes = useMemo(() => {
   176→    const set = new Set<string>();
   177→    for (const p of result.data?.products ?? []) set.add(p.partType?.name ?? "Other");
   178→    return Array.from(set).sort();
   179→  }, [result.data]);
   180→
   181→  const filteredProducts = useMemo(() => {
   182→    if (activePart === "ANY") return result.data?.products ?? [];

   184→    return (result.data?.products ?? []).filter((p) => {
   185→      const name = (p.partType?.name ?? "").toUpperCase();
   186→      if (activePart === "ALL") return true;
   187→      // Match LCD/TOUCH/BATTERY/FRAME/FLEX loosely
   188→      return name.includes(activePart);
   189→    });
   190→  }, [result.data, activePart]);
   191→
   192→  const hasQuery = query.length >= 1;
   193→  const isLoading = result.isLoading && hasQuery;
   194→  const isEmpty = hasQuery && !isLoading && (result.data?.models.length ?? 0) === 0;

   196→  return (

   199→        title="Compatibility Engine"
   200→        description="Find which phones share LCDs, touch, batteries, frames & flex"
   201→        icon={Puzzle}

   218→            className="h-16 border-0 bg-transparent pl-14 pr-28 text-base shadow-none focus-visible:ring-0 sm:text-lg"
   219→          />
   220→          {input && (
   221→            <button
   222→              onClick={() => setInput("")}
   223→              className="absolute right-4 top-1/2 -translate-y-1/2 rounded-full p-1.5 text-muted-foreground hover:bg-muted"
   224→              aria-label="Clear search"
   225→            >

   231→          <Sparkles className="h-3.5 w-3.5 text-primary" />
   232→          <span>Instant knowledge lookup · bidirectional compatibility · live inventory</span>
   233→        </div>
   234→      </Card>
   235→
   236→      {/* Knowledge queries (always visible when no results) */}
   237→      <AnimatePresence mode="wait">
   238→        {!hasQuery && (
   239→          <motion.div
   240→            key="knowledge"
   241→            initial={{ opacity: 0, y: 8 }}
   242→            animate={{ opacity: 1, y: 0 }}
   243→            exit={{ opacity: 0, y: -8 }}
   244→            className="space-y-6"
   245→          >
   246→            <div>
   247→              <div className="mb-3 flex items-center gap-2">
   248→                <CircleDot className="h-4 w-4 text-primary" />
   249→                <h3 className="text-sm font-semibold tracking-wide text-muted-foreground uppercase">
   250→                  Knowledge Queries
   251→                </h3>
   252→              </div>
   253→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   254→                {KNOWLEDGE_QUERIES.map((q) => (
   255→                  <button
   256→                    key={q.label}
   257→                    onClick={() => setInput(q.q)}
   258→                    className="group flex items-start gap-3 rounded-xl border bg-card p-4 text-left shadow-soft transition hover:border-primary/40 hover:shadow-card"
   259→                  >
   260→                    <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary transition group-hover:bg-primary/15">

   264→                      <p className="text-sm font-semibold">{q.label}</p>
   265→                      <p className="mt-0.5 text-xs text-muted-foreground line-clamp-2">{q.hint}</p>
   266→                    </div>
   267→                    <ArrowRight className="mt-1 h-4 w-4 shrink-0 text-muted-foreground transition group-hover:translate-x-0.5 group-hover:text-primary" />

   269→                ))}
   270→              </div>
   271→            </div>
   272→
   273→            {/* Hero empty state */}
   274→            <Card className="relative overflow-hidden border-dashed bg-gradient-to-br from-primary/5 via-card to-card p-8 text-center shadow-soft">
   275→              <div className="bg-grid pointer-events-none absolute inset-0 opacity-30" />
   276→              <div className="relative">
   277→                <div className="mx-auto flex h-16 w-16 items-center justify-center rounded-2xl bg-primary/10 text-primary">
   278→                  <Boxes className="h-8 w-8" />
   279→                </div>
   280→                <h3 className="mt-4 text-lg font-bold">Start typing to discover compatible parts</h3>
   281→                <p className="mx-auto mt-1 max-w-md text-sm text-muted-foreground">
   282→                  The Compatibility Engine instantly cross-references phone models — showing you every
   283→                  interchangeable LCD, touch, battery, frame, and flex cable in your inventory.
   284→                </p>
   285→                <div className="mt-5 flex flex-wrap justify-center gap-2">
   286→                  {PART_KINDS.filter((p) => p.key !== "ALL").map((p) => (
   287→                    <Badge key={p.key} variant="outline" className={cn("gap-1.5 py-1.5", p.color)}>
   288→                      <p.icon className="h-3.5 w-3.5" /> {p.label}
   289→                    </Badge>
   290→                  ))}
   291→                </div>
   292→              </div>
   293→            </Card>
   294→          </motion.div>
   295→        )}
   296→      </AnimatePresence>
   297→
   298→      {/* Loading */}
   299→      {isLoading && <CompatSkeleton />}
   300→
   301→      {/* Error */}
   302→      {result.isError && hasQuery && (
   303→        <ErrorState message={result.error.message} onRetry={() => result.refetch()} />
   304→      )}
   305→
   306→      {/* No matches */}
   307→      {isEmpty && (
   308→        <Card className="p-0">
   309→          <EmptyState

   314→              <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setManageOpen(true)}>
   315→                <Plus className="h-4 w-4" /> Add Compatibility Link
   316→              </Button>
   317→            }
   318→          />
   319→        </Card>
   320→      )}

   326→          initial={{ opacity: 0, y: 8 }}
   327→          animate={{ opacity: 1, y: 0 }}
   328→          className="space-y-6"
   329→        >
   330→          {/* Matched models */}
   331→          <section>
   332→            <SectionHeader icon={Smartphone} title="Matched Models" count={result.data?.models.length ?? 0} />
   333→            <div className="flex flex-wrap gap-2">
   334→              {result.data?.models.map((m) => (
   335→                <motion.div
   336→                  key={m.id}
   337→                  initial={{ opacity: 0, scale: 0.96 }}
   338→                  animate={{ opacity: 1, scale: 1 }}
   339→                  className="group flex items-center gap-2.5 rounded-xl border bg-card px-3.5 py-2.5 shadow-soft transition hover:border-primary/40 hover:shadow-card"
   340→                >
   341→                  <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
   342→                    <Smartphone className="h-4 w-4" />
   343→                  </div>
   344→                  <div className="min-w-0">
   345→                    <p className="text-sm font-semibold leading-tight">{m.name}</p>
   346→                    <p className="text-[11px] text-muted-foreground">
   347→                      {m.brand?.name ?? "—"}{m.releaseYear ? ` · ${m.releaseYear}` : ""}
   348→                    </p>
   349→                  </div>
   350→                </motion.div>
   351→              ))}
   352→            </div>
   353→          </section>
   354→
   355→          {/* Compatible peers grouped by part type */}
   356→          <section>
   357→            <SectionHeader icon={Link2} title="Compatible Peers" count={result.data?.peers.length ?? 0} subtitle="Phones that share parts with the matched models" />
   358→            {peersByPart.size === 0 ? (
   359→              <Card className="p-0">
   360→                <EmptyState
   361→                  icon={Link2}
   362→                  title="No compatibility links yet"
   363→                  description="Add a link to mark which other models share parts with these phones."
   364→                  action={
   365→                    <Button size="sm" variant="outline" className="gap-1.5" onClick={() => setManageOpen(true)}>
   366→                      <Plus className="h-4 w-4" /> Add Link
   367→                    </Button>
   368→                  }
   369→                />
   370→              </Card>
   371→            ) : (
   372→              <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
   373→                {PART_KINDS.filter((p) => p.key !== "ALL").map((pk) => {
   374→                  const peers = peersByPart.get(pk.key) ?? [];
   375→                  if (peers.length === 0) return null;
   376→                  const Icon = pk.icon;
   377→                  return (
   378→                    <Card key={pk.key} className="overflow-hidden p-0 shadow-soft">
   379→                      <div className={cn("flex items-center justify-between px-4 py-3", pk.color)}>
   380→                        <div className="flex items-center gap-2">
   381→                          <Icon className="h-4 w-4" />
   382→                          <span className="text-sm font-semibold">{pk.label}</span>
   383→                        </div>
   384→                        <Badge variant="outline" className="border-current/20 bg-card/40 text-foreground">
   385→                          {peers.length}
   386→                        </Badge>
   387→                      </div>
   388→                      <ScrollArea className="max-h-56">
   389→                        <div className="space-y-1 p-2.5">
   390→                          {peers.map((p) => (
   391→                            <button
   392→                              key={p.linkId}
   393→                              onClick={() => setInput(p.name)}
   394→                              className="group flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition hover:bg-muted"
   395→                            >
   396→                              <span className={cn("h-1.5 w-1.5 shrink-0 rounded-full", pk.dot)} />
   397→                              <div className="min-w-0 flex-1">
   398→                                <p className="truncate text-sm font-medium">{p.name}</p>
   399→                                <p className="text-[11px] text-muted-foreground">{p.brand ?? "—"}</p>
   400→                              </div>
   401→                              <ChevronRight className="h-3.5 w-3.5 text-muted-foreground transition group-hover:translate-x-0.5 group-hover:text-primary" />
   402→                            </button>
   403→                          ))}
   404→                        </div>
   405→                      </ScrollArea>
   406→                    </Card>
   407→                  );
   408→                })}
   409→              </div>
   410→            )}
   411→          </section>
   412→
   413→          {/* Available products */}
   414→          <section>
   415→            <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   416→              <SectionHeader
   417→                icon={Package}
   418→                title="Available Products"
   419→                count={result.data?.products.length ?? 0}
   420→                subtitle="Inventory available across matched & peer models"
   421→                noMargin
   422→              />
   423→            </div>
   424→
   425→            {/* Part-type filter chips */}
   426→            {availablePartTypes.length > 0 && (
   427→              <div className="mb-3 flex flex-wrap items-center gap-2">
   428→                <Filter className="h-3.5 w-3.5 text-muted-foreground" />
   429→                <button
   430→                  onClick={() => setActivePart("ANY")}
   431→                  className={cn(
   432→                    "rounded-full px-3 py-1.5 text-xs font-medium transition",
   433→                    activePart === "ANY"
   434→                      ? "bg-primary text-primary-foreground shadow-soft"
   435→                      : "bg-muted text-muted-foreground hover:bg-muted/70"
   436→                  )}
   437→                >
   438→                  All ({result.data?.products.length ?? 0})
   439→                </button>
   440→                {availablePartTypes.map((pt) => {
   441→                  const count = productsByPart.get(pt)?.length ?? 0;
   442→                  const upper = pt.toUpperCase();
   443→                  const pk = PART_KINDS.find((p) => upper.includes(p.key)) ?? PART_KINDS.find((p) => p.key === "ALL")!;
   444→                  const active = activePart === pk.key || (activePart === "ANY" && false);
   445→                  return (
   446→                    <button
   447→                      key={pt}
   448→                      onClick={() => setActivePart(pk.key)}
   449→                      className={cn(
   450→                        "inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium transition",
   451→                        active
   452→                          ? cn(pk.color, "ring-1", pk.ring)
   453→                          : "bg-muted text-muted-foreground hover:bg-muted/70"
   454→                      )}
   455→                    >
   456→                      <pk.icon className="h-3 w-3" />
   457→                      {pt} ({count})
   458→                    </button>
   459→                  );
   460→                })}
   461→              </div>
   462→            )}
   463→
   464→            {filteredProducts.length === 0 ? (
   465→              <Card className="p-0">
   466→                <EmptyState icon={Package} title="No products in stock" description="No inventory found for these compatible models." />
   467→              </Card>
   468→            ) : (
   469→              <CompatProductsTable products={filteredProducts} />
   470→            )}
   471→          </section>
   472→        </motion.div>
   473→      )}
   474→
   475→      {/* Manage dialog */}
   476→      <ManageCompatDialog open={manageOpen} onOpenChange={setManageOpen} />
   477→    </div>
   478→  );
   479→}

   481→// ─── Sub-components ───────────────────────────────────────────────────────────
   482→function SectionHeader({
   483→  icon: Icon, title, subtitle, count, noMargin,
   484→}: {
   485→  icon: React.ComponentType<{ className?: string }>;
   486→  title: string;
   487→  subtitle?: string;
   488→  count?: number;
   489→  noMargin?: boolean;
   490→}) {
   491→  return (
   492→    <div className={cn("flex items-center justify-between gap-3", !noMargin && "mb-3")}>
   493→      <div className="flex items-center gap-2.5">
   494→        <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
   495→          <Icon className="h-4 w-4" />
   496→        </div>
   497→        <div>
   498→          <h3 className="text-sm font-bold tracking-tight sm:text-base">{title}</h3>
   499→          {subtitle && <p className="text-xs text-muted-foreground">{subtitle}</p>}
   500→        </div>
   501→      </div>
   502→      {count !== undefined && (
   503→        <Badge variant="secondary" className="font-semibold">{count}</Badge>
   504→      )}
   505→    </div>
   506→  );
   507→}

   591→// ─── Manage compatibility dialog ──────────────────────────────────────────────
   592→function ManageCompatDialog({
   593→  open, onOpenChange,
   594→}: {
   595→  open: boolean;
   596→  onOpenChange: (o: boolean) => void;
   597→}) {
   598→  const qc = useQueryClient();
   599→  const [modelId, setModelId] = useState("");
   600→  const [peerId, setPeerId] = useState("");
   601→  const [partType, setPartType] = useState("LCD");
   602→  const [note, setNote] = useState("");
   603→  const [searchByModel, setSearchByModel] = useState("");
   604→
   605→  // Load all models (for the two dropdowns)
   606→  const models = useQuery<PhoneModel[]>({
   607→    queryKey: ["models", "manage"],
   608→    queryFn: () => api.get<PhoneModel[]>("/models"),
   609→    enabled: open,
   610→    staleTime: 60_000,
   611→  });
   612→
   613→  // Load compatibility links for the selected model (search by name)
   614→  const selectedModel = models.data?.find((m) => m.id === modelId);
   615→  const links = useQuery<CompatResult>({
   616→    queryKey: ["compat", selectedModel?.name ?? ""],
   617→    queryFn: () => api.get<CompatResult>(`/compatibility?q=${encodeURIComponent(selectedModel?.name ?? "")}`),
   618→    enabled: open && !!selectedModel?.name,
   619→    staleTime: 30_000,
   620→  });
   621→
   622→  // Filtered models for dropdown search
   623→  const filteredModelList = useMemo(() => {
   624→    const list = models.data ?? [];
   625→    if (!searchByModel) return list;
   626→    return list.filter((m) => m.name.toLowerCase().includes(searchByModel.toLowerCase()) || m.brand?.name?.toLowerCase().includes(searchByModel.toLowerCase()));
   627→  }, [models.data, searchByModel]);
   628→
   629→  const onAdd = async () => {
   630→    if (!modelId || !peerId) {
   631→      toast.error("Select both a model and a peer model");
   632→      return;
   633→    }
   634→    if (modelId === peerId) {
   635→      toast.error("A model cannot be linked to itself");
   636→      return;
   637→    }
   638→    try {
   639→      await api.post("/compatibility", { modelId, peerId, partType, note: note || null });
   640→      toast.success("Compatibility link added");
   641→      qc.invalidateQueries({ queryKey: ["compat"] });
   642→      setPeerId("");
   643→      setNote("");
   644→    } catch (e) {
   645→      toast.error((e as Error).message);
   646→    }
   647→  };
   648→
   649→  const onDelete = async (linkId: string, peerName: string, pt: string) => {
   650→    try {
   651→      await api.del(`/compatibility?id=${linkId}`);
   652→      toast.success(`Removed ${pt} link with ${peerName}`);
   653→      qc.invalidateQueries({ queryKey: ["compat"] });
   654→    } catch (e) {
   655→      toast.error((e as Error).message);
   656→    }
   657→  };
   658→
   659→  const existingLinks = links.data?.peers ?? [];
   660→
   661→  return (
   662→    <Dialog open={open} onOpenChange={onOpenChange}>
   663→      <DialogContent className="max-w-3xl gap-0 p-0">
   664→        <DialogHeader className="border-b px-6 py-4">
   665→          <DialogTitle className="flex items-center gap-2">
   666→            <Link2 className="h-4 w-4 text-primary" /> Manage Compatibility Links
   667→          </DialogTitle>
   668→          <DialogDescription>
   669→            Link two phone models that share a part type. Links are bidirectional — adding a link automatically makes both models compatible.
   670→          </DialogDescription>
   671→        </DialogHeader>
   672→
   673→        <div className="grid max-h-[70vh] grid-cols-1 overflow-hidden lg:grid-cols-[1fr_1fr]">
   674→          {/* Add form */}
   675→          <div className="space-y-4 border-b p-6 lg:border-b-0 lg:border-r">
   676→            <h3 className="text-sm font-semibold">Add New Link</h3>
   677→
   678→            <div className="space-y-1.5">
   679→              <Label className="text-xs">Model</Label>
   680→              <Select value={modelId} onValueChange={(v) => { setModelId(v); }}>
   681→                <SelectTrigger><SelectValue placeholder="Select a phone model" /></SelectTrigger>
   682→                <SelectContent>
   683→                  <div className="p-2">
   684→                    <Input
   685→                      placeholder="Search models…"
   686→                      value={searchByModel}
   687→                      onChange={(e) => setSearchByModel(e.target.value)}
   688→                      className="h-8"
   689→                    />
   690→                  </div>
   691→                  <ScrollArea className="max-h-60">
   692→                    {filteredModelList.map((m) => (
   693→                      <SelectItem key={m.id} value={m.id}>
   694→                        {m.name} <span className="text-xs text-muted-foreground">· {m.brand?.name}</span>
   695→                      </SelectItem>
   696→                    ))}
   697→                  </ScrollArea>
   698→                </SelectContent>
   699→              </Select>
   700→            </div>
   701→
   702→            <div className="space-y-1.5">
   703→              <Label className="text-xs">Peer Model</Label>
   704→              <Select value={peerId} onValueChange={setPeerId}>
   705→                <SelectTrigger><SelectValue placeholder="Select a compatible model" /></SelectTrigger>
   706→                <SelectContent>
   707→                  <ScrollArea className="max-h-60">
   708→                    {(models.data ?? [])
   709→                      .filter((m) => m.id !== modelId)
   710→                      .map((m) => (
   711→                        <SelectItem key={m.id} value={m.id}>
   712→                          {m.name} <span className="text-xs text-muted-foreground">· {m.brand?.name}</span>
   713→                        </SelectItem>
   714→                      ))}
   715→                  </ScrollArea>
   716→                </SelectContent>
   717→              </Select>
   718→            </div>
   719→
   720→            <div className="space-y-1.5">
   721→              <Label className="text-xs">Part Type</Label>
   722→              <div className="flex flex-wrap gap-1.5">
   723→                {PART_KINDS.filter((p) => p.key !== "ALL").map((pk) => {
   724→                  const active = partType === pk.key;
   725→                  const Icon = pk.icon;
   726→                  return (
   727→                    <button
   728→                      key={pk.key}
   729→                      type="button"
   730→                      onClick={() => setPartType(pk.key)}
   731→                      className={cn(
   732→                        "inline-flex items-center gap-1.5 rounded-lg border px-2.5 py-1.5 text-xs font-medium transition",
   733→                        active ? cn(pk.color, "ring-1", pk.ring, "border-transparent") : "border-border hover:bg-muted"
   734→                      )}
   735→                    >
   736→                      <Icon className="h-3.5 w-3.5" /> {pk.label}
   737→                    </button>
   738→                  );
   739→                })}
   740→              </div>
   741→            </div>
   742→
   743→            <div className="space-y-1.5">
   744→              <Label className="text-xs">Note (optional)</Label>
   745→              <Input value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. Same LCD manufacturer" />
   746→            </div>
   747→
   748→            <Button className="w-full gap-1.5" onClick={onAdd} disabled={!modelId || !peerId}>
   749→              <Plus className="h-4 w-4" /> Add Link
   750→            </Button>
   751→          </div>
   752→
   753→          {/* Existing links */}
   754→          <div className="flex flex-col overflow-hidden">
   755→            <div className="flex items-center justify-between border-b px-6 py-4">
   756→              <h3 className="text-sm font-semibold">Existing Links</h3>
   757→              {selectedModel && (
   758→                <Badge variant="outline" className="max-w-[200px] truncate">
   759→                  {selectedModel.name}
   760→                </Badge>
   761→              )}
   762→            </div>
   763→            <ScrollArea className="max-h-[55vh]">
   764→              {!selectedModel ? (
   765→                <div className="px-6 py-12 text-center text-sm text-muted-foreground">
   766→                  <Smartphone className="mx-auto mb-2 h-8 w-8 opacity-40" />
   767→                  Select a model to view its existing links.
   768→                </div>
   769→              ) : links.isLoading ? (
   770→                <div className="space-y-2 p-4">
   771→                  {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-14 w-full rounded-lg" />)}
   772→                </div>
   773→              ) : existingLinks.length === 0 ? (
   774→                <div className="px-6 py-12 text-center text-sm text-muted-foreground">
   775→                  <Link2 className="mx-auto mb-2 h-8 w-8 opacity-40" />
   776→                  No compatibility links for this model yet.
   777→                </div>
   778→              ) : (
   779→                <div className="space-y-1.5 p-3">
   780→                  {existingLinks.map((l) => {
   781→                    const pk = partKindMeta(l.partType);
   782→                    const Icon = pk.icon;
   783→                    return (
   784→                      <div
   785→                        key={l.linkId}
   786→                        className="group flex items-center gap-3 rounded-lg border bg-card px-3 py-2.5 transition hover:border-primary/30"
   787→                      >
   788→                        <div className={cn("flex h-9 w-9 shrink-0 items-center justify-center rounded-lg", pk.color)}>
   789→                          <Icon className="h-4 w-4" />
   790→                        </div>
   791→                        <div className="min-w-0 flex-1">
   792→                          <p className="truncate text-sm font-medium">{l.name}</p>
   793→                          <p className="text-[11px] text-muted-foreground">
   794→                            {l.brand ?? "—"} · {pk.label}
   795→                          </p>
   796→                        </div>
   797→                        <Button
   798→                          size="icon"
   799→                          variant="ghost"
   800→                          className="h-8 w-8 text-muted-foreground hover:text-destructive"
   801→                          onClick={() => onDelete(l.linkId, l.name, pk.label)}
   802→                          aria-label={`Delete link with ${l.name}`}
   803→                        >
   804→                          <Trash2 className="h-4 w-4" />
   805→                        </Button>
   806→                      </div>
   807→                    );
   808→                  })}
   809→                </div>
   810→              )}
   811→            </ScrollArea>
   812→          </div>
   813→        </div>
   814→
   815→        <DialogFooter className="border-t px-6 py-3">
   816→          <Button variant="outline" onClick={() => onOpenChange(false)}>Close</Button>
   817→        </DialogFooter>
   818→      </DialogContent>
   819→    </Dialog>
   820→  );
   821→}
