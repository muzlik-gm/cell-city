    34→// Types
    35→// ────────────────────────────────────────────────────────────────────────────
    36→interface Customer { id: string; name: string; phone?: string | null; company?: string | null; }
    37→interface Product {
    38→  id: string; sku: string; name: string; sellingPrice: number; purchasePrice: number;
    39→  stock: number; brand?: { name: string } | null; model?: { name: string } | null;
    40→}
    41→interface SaleItem {
    42→  id: string; productId: string; name: string; qty: number; price: number;
    43→  cost: number; discount: number; total: number;
    44→  product?: { sku?: string; brand?: { name: string } | null; model?: { name: string } | null };
    45→}
    46→interface Sale {
    47→  id: string; invoiceNo: string; customerId?: string | null;
    48→  customer?: Customer | null; user?: { name: string } | null;
    49→  subtotal: number; discount: number; tax: number; total: number; profit: number;
    50→  paid: number; paymentMethod: string; paymentStatus: string; status: string;
    51→  notes?: string | null; createdAt: string; items: SaleItem[];
    52→  business?: { name: string; phone: string; address: string; email: string; currencySymbol: string; taxName: string };
    53→}
    54→interface CartLine {
    55→  key: string; productId: string; name: string; sku: string;
    56→  qty: number; price: number; cost: number; discount: number; stock: number;
    57→}

    62→export function SalesView() {
    63→  const qc = useQueryClient();
    64→  const [q, setQ] = useState("");
    65→  const [paymentStatus, setPaymentStatus] = useState("");
    66→  const [dateFilter, setDateFilter] = useState("");
    67→  const [page, setPage] = useState(1);
    68→  const [formOpen, setFormOpen] = useState(false);
    69→  const [invoiceSale, setInvoiceSale] = useState<Sale | null>(null);
    70→
    71→  const queryStr = useMemo(() => {
    72→    const p = new URLSearchParams();
    73→    if (q) p.set("q", q);
    74→    if (paymentStatus) p.set("paymentStatus", paymentStatus);
    75→    if (dateFilter) {
    76→      const d = new Date(dateFilter);
    77→      const from = new Date(d.getFullYear(), d.getMonth(), d.getDate()).toISOString();
    78→      const to = new Date(d.getFullYear(), d.getMonth(), d.getDate(), 23, 59, 59).toISOString();
    79→      p.set("from", from);
    80→      p.set("to", to);
    81→    }
    82→    p.set("page", String(page));
    83→    p.set("pageSize", "20");
    84→    return p.toString();
    85→  }, [q, paymentStatus, dateFilter, page]);
    86→
    87→  const sales = useQuery<{ data: Sale[]; total: number }>({
    88→    queryKey: ["sales", queryStr],
    89→    queryFn: () => api.get(`/sales?${queryStr}`),
    90→  });
    91→
    92→  // Stats — fetch all sales this month for aggregation
    93→  const statsQuery = useQuery<{ data: Sale[] }>({
    94→    queryKey: ["sales-stats"],
    95→    queryFn: () => api.get(`/sales?pageSize=100`),
    96→    staleTime: 30_000,
    97→  });
    98→  const stats = useMemo(() => {
    99→    const all = statsQuery.data?.data ?? [];
   100→    const now = new Date();
   101→    const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   102→    const monthStart = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
   103→    let todayTotal = 0, todayCount = 0, monthTotal = 0, outstanding = 0;
   104→    for (const s of all) {
   105→      const t = new Date(s.createdAt).getTime();
   106→      if (s.status !== "RETURNED") {
   107→        if (t >= todayStart) { todayTotal += s.total; todayCount++; }
   108→        if (t >= monthStart) monthTotal += s.total;
   109→      }
   110→      if (s.paymentStatus !== "PAID") outstanding += Math.max(0, s.total - s.paid);
   111→    }
   112→    return { todayTotal, todayCount, monthTotal, outstanding };
   113→  }, [statsQuery.data]);
   114→
   115→  const columns: Column<Sale & Record<string, unknown>>[] = [
   116→    {
   117→      key: "invoiceNo", header: "Invoice", className: "min-w-[140px]",
   118→      render: (s) => (
   119→        <div>
   120→          <p className="font-mono text-xs font-semibold text-foreground">{s.invoiceNo}</p>
   121→          <p className="text-[11px] text-muted-foreground">{formatDateTime(s.createdAt)}</p>
   122→        </div>
   123→      ),
   124→    },
   125→    {
   126→      key: "customer", header: "Customer", className: "min-w-[160px]",
   127→      render: (s) => (
   128→        <div className="flex items-center gap-2">
   129→          <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary">
   130→            <User2 className="h-4 w-4" />
   131→          </div>
   132→          <div className="min-w-0">
   133→            <p className="truncate text-sm font-medium">{s.customer?.name ?? "Walk-in Customer"}</p>
   134→            <p className="text-[11px] text-muted-foreground">{s.customer?.phone ?? "—"}</p>
   135→          </div>
   136→        </div>
   137→      ),
   138→    },
   139→    {
   140→      key: "items", header: "Items",
   141→      render: (s) => <Badge variant="secondary" className="font-semibold">{s.items?.length ?? 0}</Badge>,
   142→    },
   143→    {
   144→      key: "total", header: "Total", className: "text-right",
   145→      render: (s) => (
   146→        <div className="text-right">
   147→          <p className="text-sm font-bold">{formatCurrency(s.total)}</p>
   148→          <p className="text-[11px] text-muted-foreground">profit {formatCurrency(s.profit)}</p>
   149→        </div>
   150→      ),
   151→    },
   152→    {
   153→      key: "paymentMethod", header: "Method",
   154→      render: (s) => <PaymentMethodBadge method={s.paymentMethod} />,
   155→    },
   156→    {
   157→      key: "paymentStatus", header: "Payment",
   158→      render: (s) => <PaymentStatusBadge status={s.paymentStatus} />,
   159→    },
   160→    {
   161→      key: "status", header: "Status",
   162→      render: (s) => (
   163→        s.status === "RETURNED" ? (
   164→          <Badge variant="outline" className="bg-rose-500/10 text-rose-600 dark:text-rose-400">Returned</Badge>
   165→        ) : (
   166→          <Badge variant="outline" className="bg-emerald-500/10 text-emerald-600 dark:text-emerald-400">Completed</Badge>
   167→        )
   168→      ),
   169→    },
   170→    {
   171→      key: "actions", header: "", className: "text-right",
   172→      render: (s) => (
   173→        <div className="flex justify-end gap-1">
   174→          <Button size="sm" variant="ghost" className="h-8 gap-1" onClick={(e) => { e.stopPropagation(); openInvoice(s.id); }}>
   175→            <Eye className="h-3.5 w-3.5" /> View
   176→          </Button>
   177→          {s.status !== "RETURNED" && (
   178→            <Button size="sm" variant="ghost" className="h-8 gap-1 text-amber-600" onClick={(e) => { e.stopPropagation(); handleReturn(s); }}>
   179→              <Undo2 className="h-3.5 w-3.5" /> Return
   180→            </Button>
   181→          )}
   182→        </div>
   183→      ),
   184→    },
   185→  ];
   186→
   187→  const openInvoice = async (id: string) => {
   188→    try {
   189→      const sale = await api.get<Sale>(`/sales/${id}`);
   190→      setInvoiceSale(sale);
   191→    } catch (e) {
   192→      toast.error((e as Error).message);
   193→    }
   194→  };
   195→
   196→  const handleReturn = async (s: Sale) => {
   197→    if (!confirm(`Mark invoice ${s.invoiceNo} as RETURNED? Items will be restocked.`)) return;
   198→    try {
   199→      await api.put(`/sales/${s.id}`, { status: "RETURNED" });
   200→      qc.invalidateQueries({ queryKey: ["sales"] });
   201→      qc.invalidateQueries({ queryKey: ["products"] });
   202→      toast.success(`Invoice ${s.invoiceNo} marked as returned. Stock restocked.`);
   203→    } catch (e) {
   204→      toast.error((e as Error).message);
   205→    }
   206→  };
   207→
   208→  const hasFilters = q || paymentStatus || dateFilter;
   209→  const clearFilters = () => { setQ(""); setPaymentStatus(""); setDateFilter(""); setPage(1); };
   210→
   211→  return (
   212→    <div className="space-y-5">
   213→      <PageHeader
   214→        title="Sales & Invoices"
   215→        description="Create POS sales, manage invoices, track payments and returns"
   216→        icon={ShoppingCart}
   217→        actions={
   218→          <Button size="sm" className="gap-1.5" onClick={() => setFormOpen(true)}>
   219→            <Plus className="h-4 w-4" /> New Sale
   220→          </Button>
   221→        }
   222→      />
   223→
   224→      {/* Stats */}
   225→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   226→        <StatCard label="Today's Sales" value={formatCurrency(stats.todayTotal)} icon={Wallet} accent="emerald" subtitle={`${stats.todayCount} invoices today`} />
   227→        <StatCard label="Today's Count" value={stats.todayCount} icon={ShoppingCart} accent="teal" subtitle="Invoices issued today" />
   228→        <StatCard label="This Month" value={formatCurrency(stats.monthTotal)} icon={TrendingUp} accent="purple" subtitle="Total revenue this month" />
   229→        <StatCard label="Outstanding" value={formatCurrency(stats.outstanding)} icon={AlertCircle} accent="amber" subtitle="Unpaid invoices" />
   230→      </div>
   231→
   232→      {/* Filters */}
   233→      <Card className="p-4 shadow-soft">
   234→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center">
   235→          <div className="relative flex-1">
   236→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   237→            <Input
   238→              value={q}
   239→              onChange={(e) => { setQ(e.target.value); setPage(1); }}
   240→              placeholder="Search invoice no, customer name or phone…"
   241→              className="pl-9"
   242→            />
   243→          </div>
   244→          <div className="flex flex-wrap gap-2">
   245→            <Select value={paymentStatus || "all"} onValueChange={(v) => { setPaymentStatus(v === "all" ? "" : v); setPage(1); }}>
   246→              <SelectTrigger className="w-[150px]"><SelectValue placeholder="All payments" /></SelectTrigger>
   247→              <SelectContent>
   248→                <SelectItem value="all">All payments</SelectItem>
   249→                {PAYMENT_STATUSES.map((s) => <SelectItem key={s} value={s}>{s.charAt(0) + s.slice(1).toLowerCase()}</SelectItem>)}
   250→              </SelectContent>
   251→            </Select>
   252→            <Input
   253→              type="date"
   254→              value={dateFilter}
   255→              onChange={(e) => { setDateFilter(e.target.value); setPage(1); }}
   256→              className="w-[150px]"
   257→            />
   258→            {hasFilters && (
   259→              <Button variant="ghost" size="sm" className="gap-1" onClick={clearFilters}>
   260→                <X className="h-4 w-4" /> Clear
   261→              </Button>
   262→            )}
   263→          </div>
   264→        </div>
   265→      </Card>
   266→
   267→      {/* Table */}
   268→      <DataTable
   269→        columns={columns}
   270→        data={sales.data?.data ?? []}
   271→        loading={sales.isLoading}
   272→        pagination
   273→        page={page}
   274→        pageSize={20}
   275→        total={sales.data?.total ?? 0}
   276→        onPageChange={setPage}
   277→        onRowClick={(s) => openInvoice(s.id)}
   278→        rowKey={(s) => s.id}
   279→        emptyTitle="No sales found"
   280→        emptyDescription={hasFilters ? "Try adjusting your filters." : "Create your first sale to get started."}
   281→      />
   282→
   283→      <SaleFormDialog
   284→        open={formOpen}
   285→        onOpenChange={setFormOpen}
   286→        onCreated={(sale) => { setInvoiceSale(sale); qc.invalidateQueries({ queryKey: ["sales"] }); qc.invalidateQueries({ queryKey: ["products"] }); }}
   287→      />
   288→
   289→      <InvoiceDialog sale={invoiceSale} onOpenChange={(o) => !o && setInvoiceSale(null)} />
   290→    </div>
   291→  );
   292→}

   294→// ────────────────────────────────────────────────────────────────────────────
   295→// Sale Form Dialog (POS-style)
   296→// ────────────────────────────────────────────────────────────────────────────
   297→function SaleFormDialog({
   298→  open, onOpenChange, onCreated,
   299→}: {
   300→  open: boolean;
   301→  onOpenChange: (o: boolean) => void;
   302→  onCreated: (sale: Sale) => void;
   303→}) {
   304→  const qc = useQueryClient();
   305→  const [customerId, setCustomerId] = useState("");
   306→  const [cart, setCart] = useState<CartLine[]>([]);
   307→  const [search, setSearch] = useState("");
   308→  const [overallDiscount, setOverallDiscount] = useState(0);
   309→  const [tax, setTax] = useState(0);
   310→  const [paymentMethod, setPaymentMethod] = useState("CASH");
   311→  const [paymentStatus, setPaymentStatus] = useState("PAID");
   312→  const [notes, setNotes] = useState("");
   313→
   314→  const customers = useQuery<Customer[]>({
   315→    queryKey: ["customers"],
   316→    queryFn: () => api.get("/customers"),
   317→    staleTime: 60_000,
   318→  });
   319→
   320→  const productSearch = useQuery<{ data: Product[] }>({
   321→    queryKey: ["products-pos", search],
   322→    queryFn: () => api.get(`/products?q=${encodeURIComponent(search)}&pageSize=10`),
   323→    enabled: open && search.length > 0,
   324→    staleTime: 10_000,
   325→  });
   326→
   327→  const resetForm = () => {
   328→    setCustomerId(""); setCart([]); setSearch(""); setOverallDiscount(0);
   329→    setTax(0); setPaymentMethod("CASH"); setPaymentStatus("PAID"); setNotes("");
   330→  };
   331→  const handleOpenChange = (o: boolean) => {
   332→    if (!o) resetForm();
   333→    onOpenChange(o);
   334→  };
   335→
   336→  const totals = useMemo(() => {
   337→    const subtotal = cart.reduce((s, l) => s + l.qty * l.price, 0);
   338→    const lineDiscounts = cart.reduce((s, l) => s + l.discount, 0);
   339→    const cost = cart.reduce((s, l) => s + l.qty * l.cost, 0);
   340→    const net = subtotal - lineDiscounts - overallDiscount;
   341→    const total = Math.max(0, net + tax);
   342→    const profit = net - cost;
   343→    return { subtotal, lineDiscounts, cost, net, total, profit };
   344→  }, [cart, overallDiscount, tax]);
   345→
   346→  const addToCart = (p: Product) => {
   347→    setCart((prev) => {
   348→      const existing = prev.find((l) => l.productId === p.id);
   349→      if (existing) {
   350→        return prev.map((l) => l.productId === p.id ? { ...l, qty: l.qty + 1 } : l);
   351→      }
   352→      return [...prev, {
   353→        key: p.id, productId: p.id, name: p.name, sku: p.sku,
   354→        qty: 1, price: p.sellingPrice, cost: p.purchasePrice, discount: 0, stock: p.stock,
   355→      }];
   356→    });
   357→    setSearch("");
   358→  };
   359→
   360→  const updateLine = (key: string, patch: Partial<CartLine>) => {
   361→    setCart((prev) => prev.map((l) => l.key === key ? { ...l, ...patch } : l));
   362→  };
   363→  const removeLine = (key: string) => setCart((prev) => prev.filter((l) => l.key !== key));
   364→
   365→  const createSale = useMutation({
   366→    mutationFn: () => api.post<Sale>("/sales", {
   367→      customerId: customerId || undefined,
   368→      items: cart.map((l) => ({ productId: l.productId, qty: l.qty, price: l.price, discount: l.discount })),
   369→      discount: overallDiscount,
   370→      tax,
   371→      paymentMethod,
   372→      paymentStatus,
   373→      notes,
   374→    }),
   375→    onSuccess: (sale) => {
   376→      toast.success(`Sale ${sale.invoiceNo} created`);
   377→      qc.invalidateQueries({ queryKey: ["sales"] });
   378→      qc.invalidateQueries({ queryKey: ["products"] });
   379→      onOpenChange(false);
   380→      onCreated(sale);
   381→    },
   382→    onError: (e: Error) => toast.error(e.message),
   383→  });
   384→
   385→  const canSubmit = cart.length > 0 && !createSale.isPending;
   386→
   387→  return (
   388→    <Dialog open={open} onOpenChange={handleOpenChange}>
   389→      <DialogContent className="sm:max-w-4xl max-h-[92vh] overflow-hidden p-0 gap-0">
   390→        <DialogHeader className="px-6 py-4 border-b bg-muted/30">
   391→          <DialogTitle className="flex items-center gap-2">
   392→            <ShoppingCart className="h-5 w-5 text-primary" /> New Sale
   393→          </DialogTitle>
   394→          <DialogDescription>Add products to cart and checkout. Stock is deducted automatically.</DialogDescription>
   395→        </DialogHeader>
   396→
   397→        <div className="grid grid-cols-1 lg:grid-cols-[1fr_360px] overflow-hidden">
   398→          {/* Left: product search + cart */}
   399→          <div className="flex flex-col overflow-hidden border-r">
   400→            {/* Search */}
   401→            <div className="p-4 border-b">
   402→              <div className="relative">
   403→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   404→                <Input
   405→                  value={search}
   406→                  onChange={(e) => setSearch(e.target.value)}
   407→                  placeholder="Search products by name, SKU, barcode…"
   408→                  className="pl-9"
   409→                  autoFocus
   410→                />
   411→              </div>
   412→              {/* Search results */}
   413→              {search && (
   414→                <div className="mt-2 max-h-64 overflow-y-auto rounded-lg border bg-popover shadow-sm">
   415→                  {productSearch.isLoading && (
   416→                    <div className="p-3 text-xs text-muted-foreground flex items-center gap-2">
   417→                      <Loader2 className="h-3 w-3 animate-spin" /> Searching…
   418→                    </div>
   419→                  )}
   420→                  {!productSearch.isLoading && (productSearch.data?.data ?? []).length === 0 && (
   421→                    <div className="p-3 text-xs text-muted-foreground">No products found.</div>
   422→                  )}
   423→                  <AnimatePresence>
   424→                    {(productSearch.data?.data ?? []).map((p) => (
   425→                      <motion.button
   426→                        key={p.id}
   427→                        initial={{ opacity: 0, y: -4 }}
   428→                        animate={{ opacity: 1, y: 0 }}
   429→                        exit={{ opacity: 0 }}
   430→                        onClick={() => addToCart(p)}
   431→                        className="flex w-full items-center gap-3 border-b px-3 py-2 text-left last:border-b-0 hover:bg-accent transition"
   432→                      >
   433→                        <div className="flex h-9 w-9 items-center justify-center rounded-md bg-primary/10 text-primary">
   434→                          <Package className="h-4 w-4" />
   435→                        </div>
   436→                        <div className="min-w-0 flex-1">
   437→                          <p className="truncate text-sm font-medium">{p.name}</p>
   438→                          <p className="text-[11px] text-muted-foreground">
   439→                            {p.sku} · {p.brand?.name ?? ""} {p.model?.name ?? ""}
   440→                          </p>
   441→                        </div>
   442→                        <div className="text-right">
   443→                          <p className="text-sm font-semibold text-emerald-600">{formatCurrency(p.sellingPrice)}</p>
   444→                          <p className="text-[11px] text-muted-foreground">{p.stock} in stock</p>
   445→                        </div>
   446→                        <Plus className="h-4 w-4 text-muted-foreground" />
   447→                      </motion.button>
   448→                    ))}
   449→                  </AnimatePresence>
   450→                </div>
   451→              )}
   452→            </div>
   453→
   454→            {/* Cart list */}
   455→            <div className="flex-1 overflow-hidden">
   456→              {cart.length === 0 ? (
   457→                <div className="flex h-full flex-col items-center justify-center px-6 py-12 text-center">
   458→                  <div className="flex h-14 w-14 items-center justify-center rounded-2xl bg-muted text-muted-foreground">
   459→                    <ShoppingCart className="h-6 w-6" />
   460→                  </div>
   461→                  <h3 className="mt-3 text-sm font-semibold">Cart is empty</h3>
   462→                  <p className="mt-1 text-xs text-muted-foreground">Search products above and add them to the cart.</p>
   463→                </div>
   464→              ) : (
   465→                <ScrollArea className="h-full max-h-[42vh]">
   466→                  <div className="space-y-2 p-3">
   467→                    <AnimatePresence initial={false}>
   468→                      {cart.map((l) => (
   469→                        <motion.div
   470→                          key={l.key}
   471→                          layout
   472→                          initial={{ opacity: 0, y: -4 }}
   473→                          animate={{ opacity: 1, y: 0 }}
   474→                          exit={{ opacity: 0, x: -10 }}
   475→                          className="rounded-lg border bg-card p-3"
   476→                        >
   477→                          <div className="flex items-start gap-2">
   478→                            <div className="min-w-0 flex-1">
   479→                              <p className="truncate text-sm font-medium">{l.name}</p>
   480→                              <p className="text-[11px] text-muted-foreground">

   879→                    <span className="text-xs font-medium">Balance Due</span>
   880→                    <span className="font-semibold">{formatCurrency(sale.total - sale.paid, sale.business?.currencySymbol ?? "Rs")}</span>
   881→                  </div>
   882→                )}
   883→                <div className="flex items-center justify-between pt-1 text-xs">
   884→                  <span className="text-muted-foreground">Est. profit</span>
   885→                  <span className={`font-semibold ${sale.profit >= 0 ? "text-emerald-600" : "text-rose-600"}`}>
   886→                    {formatCurrency(sale.profit, sale.business?.currencySymbol ?? "Rs")}
   887→                  </span>
   888→                </div>
   889→              </div>
   890→
   891→              {sale.notes && (
   892→                <div className="rounded-lg border border-amber-500/20 bg-amber-500/5 p-3">
   893→                  <p className="mb-1 text-[11px] font-semibold uppercase text-amber-600">Notes</p>
   894→                  <p className="text-sm">{sale.notes}</p>
   895→                </div>
   896→              )}
   897→
   898→              {sale.user && (
   899→                <p className="text-center text-[11px] text-muted-foreground">Served by {sale.user.name}</p>
   900→              )}
   901→            </div>
   902→          </ScrollArea>
   903→        )}
   904→      </DialogContent>
   905→    </Dialog>
   906→  );
   907→}
