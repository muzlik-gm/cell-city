    35→// Types
    36→// ────────────────────────────────────────────────────────────────────────────
    37→interface Customer { id: string; name: string; phone?: string | null; company?: string | null; }
    38→interface Product {
    39→  id: string; sku: string; name: string; sellingPrice: number; purchasePrice: number;
    40→  stock: number; brand?: { name: string } | null; model?: { name: string } | null;
    41→}
    42→interface SaleItem {
    43→  id: string; productId: string; name: string; qty: number; price: number;
    44→  cost: number; discount: number; total: number;
    45→  product?: { sku?: string; brand?: { name: string } | null; model?: { name: string } | null };
    46→}
    47→interface Sale {
    48→  id: string; invoiceNo: string; customerId?: string | null;
    49→  customer?: Customer | null; user?: { name: string } | null;
    50→  subtotal: number; discount: number; tax: number; total: number; profit: number;
    51→  paid: number; paymentMethod: string; paymentStatus: string; status: string;
    52→  notes?: string | null; createdAt: string; items: SaleItem[];
    53→  business?: { name: string; phone: string; address: string; email: string; currencySymbol: string; taxName: string };
    54→}
    55→interface CartLine {
    56→  key: string; productId: string; name: string; sku: string;
    57→  qty: number; price: number; cost: number; discount: number; stock: number;
    58→}

    63→export function SalesView() {
    64→  const qc = useQueryClient();
    65→  const [q, setQ] = useState("");
    66→  const [paymentStatus, setPaymentStatus] = useState("");
    67→  const [dateFilter, setDateFilter] = useState("");
    68→  const [page, setPage] = useState(1);
    69→  const [formOpen, setFormOpen] = useState(false);
    70→  const [invoiceSale, setInvoiceSale] = useState<Sale | null>(null);
    71→
    72→  const queryStr = useMemo(() => {
    73→    const p = new URLSearchParams();
    74→    if (q) p.set("q", q);
    75→    if (paymentStatus) p.set("paymentStatus", paymentStatus);
    76→    if (dateFilter) {
    77→      const d = new Date(dateFilter);
    78→      const from = new Date(d.getFullYear(), d.getMonth(), d.getDate()).toISOString();
    79→      const to = new Date(d.getFullYear(), d.getMonth(), d.getDate(), 23, 59, 59).toISOString();
    80→      p.set("from", from);
    81→      p.set("to", to);
    82→    }
    83→    p.set("page", String(page));
    84→    p.set("pageSize", "20");
    85→    return p.toString();
    86→  }, [q, paymentStatus, dateFilter, page]);
    87→
    88→  const sales = useQuery<{ data: Sale[]; total: number }>({
    89→    queryKey: ["sales", queryStr],
    90→    queryFn: () => api.get(`/sales?${queryStr}`),
    91→  });
    92→
    93→  // Stats — fetch all sales this month for aggregation
    94→  const statsQuery = useQuery<{ data: Sale[] }>({
    95→    queryKey: ["sales-stats"],
    96→    queryFn: () => api.get(`/sales?pageSize=100`),
    97→    staleTime: 30_000,
    98→  });
    99→  const stats = useMemo(() => {
   100→    const all = statsQuery.data?.data ?? [];
   101→    const now = new Date();
   102→    const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   103→    const monthStart = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
   104→    let todayTotal = 0, todayCount = 0, monthTotal = 0, outstanding = 0;
   105→    for (const s of all) {
   106→      const t = new Date(s.createdAt).getTime();
   107→      if (s.status !== "RETURNED") {
   108→        if (t >= todayStart) { todayTotal += s.total; todayCount++; }
   109→        if (t >= monthStart) monthTotal += s.total;
   110→      }
   111→      if (s.paymentStatus !== "PAID") outstanding += Math.max(0, s.total - s.paid);
   112→    }
   113→    return { todayTotal, todayCount, monthTotal, outstanding };
   114→  }, [statsQuery.data]);
   115→
   116→  const columns: Column<Sale & Record<string, unknown>>[] = [
   117→    {
   118→      key: "invoiceNo", header: "Invoice", className: "min-w-[140px]",
   119→      render: (s) => (
   120→        <div>
   121→          <p className="font-mono text-xs font-semibold text-foreground">{s.invoiceNo}</p>
   122→          <p className="text-[11px] text-muted-foreground">{formatDateTime(s.createdAt)}</p>
   123→        </div>
   124→      ),
   125→    },
   126→    {
   127→      key: "customer", header: "Customer", className: "min-w-[160px]",
   128→      render: (s) => (
   129→        <div className="flex items-center gap-2">
   130→          <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary">
   131→            <User2 className="h-4 w-4" />
   132→          </div>
   133→          <div className="min-w-0">
   134→            <p className="truncate text-sm font-medium">{s.customer?.name ?? "Walk-in Customer"}</p>
   135→            <p className="text-[11px] text-muted-foreground">{s.customer?.phone ?? "—"}</p>
   136→          </div>
   137→        </div>
   138→      ),
   139→    },
   140→    {
   141→      key: "items", header: "Items",
   142→      render: (s) => <Badge variant="secondary" className="font-semibold">{s.items?.length ?? 0}</Badge>,
   143→    },
   144→    {
   145→      key: "total", header: "Total", className: "text-right",
   146→      render: (s) => (
   147→        <div className="text-right">
   148→          <p className="text-sm font-bold">{formatCurrency(s.total)}</p>
   149→          <p className="text-[11px] text-muted-foreground">profit {formatCurrency(s.profit)}</p>
   150→        </div>
   151→      ),
   152→    },
   153→    {
   154→      key: "paymentMethod", header: "Method",
   155→      render: (s) => <PaymentMethodBadge method={s.paymentMethod} />,
   156→    },
   157→    {
   158→      key: "paymentStatus", header: "Payment",
   159→      render: (s) => <PaymentStatusBadge status={s.paymentStatus} />,
   160→    },
   161→    {
   162→      key: "status", header: "Status",
   163→      render: (s) => (
   164→        s.status === "RETURNED" ? (
   165→          <Badge variant="outline" className="bg-rose-500/10 text-rose-600 dark:text-rose-400">Returned</Badge>
   166→        ) : (
   167→          <Badge variant="outline" className="bg-emerald-500/10 text-emerald-600 dark:text-emerald-400">Completed</Badge>
   168→        )
   169→      ),
   170→    },
   171→    {
   172→      key: "actions", header: "", className: "text-right",
   173→      render: (s) => (
   174→        <div className="flex justify-end gap-1">
   175→          <Button size="sm" variant="ghost" className="h-8 gap-1" onClick={(e) => { e.stopPropagation(); openInvoice(s.id); }}>
   176→            <Eye className="h-3.5 w-3.5" /> View
   177→          </Button>
   178→          {s.status !== "RETURNED" && (
   179→            <Button size="sm" variant="ghost" className="h-8 gap-1 text-amber-600" onClick={(e) => { e.stopPropagation(); handleReturn(s); }}>
   180→              <Undo2 className="h-3.5 w-3.5" /> Return
   181→            </Button>
   182→          )}
   183→        </div>
   184→      ),
   185→    },
   186→  ];
   187→
   188→  const openInvoice = async (id: string) => {
   189→    try {
   190→      const sale = await api.get<Sale>(`/sales/${id}`);
   191→      setInvoiceSale(sale);
   192→    } catch (e) {
   193→      toast.error((e as Error).message);
   194→    }
   195→  };
   196→
   197→  const handleReturn = async (s: Sale) => {
   198→    if (!confirm(`Mark invoice ${s.invoiceNo} as RETURNED? Items will be restocked.`)) return;
   199→    try {
   200→      await api.put(`/sales/${s.id}`, { status: "RETURNED" });
   201→      qc.invalidateQueries({ queryKey: ["sales"] });
   202→      qc.invalidateQueries({ queryKey: ["products"] });
   203→      toast.success(`Invoice ${s.invoiceNo} marked as returned. Stock restocked.`);
   204→    } catch (e) {
   205→      toast.error((e as Error).message);
   206→    }
   207→  };
   208→
   209→  const hasFilters = q || paymentStatus || dateFilter;
   210→  const clearFilters = () => { setQ(""); setPaymentStatus(""); setDateFilter(""); setPage(1); };
   211→
   212→  return (
   213→    <div className="space-y-5">
   214→      <PageHeader
   215→        title="Sales & Invoices"
   216→        description="Create POS sales, manage invoices, track payments and returns"
   217→        icon={ShoppingCart}
   218→        actions={
   219→          <Button size="sm" className="gap-1.5" onClick={() => setFormOpen(true)}>
   220→            <Plus className="h-4 w-4" /> New Sale
   221→          </Button>
   222→        }
   223→      />
   224→
   225→      {/* Stats */}
   226→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   227→        <StatCard label="Today's Sales" value={formatCurrency(stats.todayTotal)} icon={Wallet} accent="emerald" subtitle={`${stats.todayCount} invoices today`} />
   228→        <StatCard label="Today's Count" value={stats.todayCount} icon={ShoppingCart} accent="teal" subtitle="Invoices issued today" />
   229→        <StatCard label="This Month" value={formatCurrency(stats.monthTotal)} icon={TrendingUp} accent="purple" subtitle="Total revenue this month" />
   230→        <StatCard label="Outstanding" value={formatCurrency(stats.outstanding)} icon={AlertCircle} accent="amber" subtitle="Unpaid invoices" />
   231→      </div>
   232→
   233→      {/* Filters */}
   234→      <Card className="p-4 shadow-soft">
   235→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center">
   236→          <div className="relative flex-1">
   237→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   238→            <Input
   239→              value={q}
   240→              onChange={(e) => { setQ(e.target.value); setPage(1); }}
   241→              placeholder="Search invoice no, customer name or phone…"
   242→              className="pl-9"
   243→            />
   244→          </div>
   245→          <div className="flex flex-wrap gap-2">
   246→            <Select value={paymentStatus || "all"} onValueChange={(v) => { setPaymentStatus(v === "all" ? "" : v); setPage(1); }}>
   247→              <SelectTrigger className="w-[150px]"><SelectValue placeholder="All payments" /></SelectTrigger>
   248→              <SelectContent>
   249→                <SelectItem value="all">All payments</SelectItem>
   250→                {PAYMENT_STATUSES.map((s) => <SelectItem key={s} value={s}>{s.charAt(0) + s.slice(1).toLowerCase()}</SelectItem>)}
   251→              </SelectContent>
   252→            </Select>
   253→            <Input
   254→              type="date"
   255→              value={dateFilter}
   256→              onChange={(e) => { setDateFilter(e.target.value); setPage(1); }}
   257→              className="w-[150px]"
   258→            />
   259→            {hasFilters && (
   260→              <Button variant="ghost" size="sm" className="gap-1" onClick={clearFilters}>
   261→                <X className="h-4 w-4" /> Clear
   262→              </Button>
   263→            )}
   264→          </div>
   265→        </div>
   266→      </Card>
   267→
   268→      {/* Table */}
   269→      <DataTable
   270→        columns={columns}
   271→        data={sales.data?.data ?? []}
   272→        loading={sales.isLoading}
   273→        pagination
   274→        page={page}
   275→        pageSize={20}
   276→        total={sales.data?.total ?? 0}
   277→        onPageChange={setPage}
   278→        onRowClick={(s) => openInvoice(s.id)}
   279→        rowKey={(s) => s.id}
   280→        emptyTitle="No sales found"
   281→        emptyDescription={hasFilters ? "Try adjusting your filters." : "Create your first sale to get started."}
   282→      />
   283→
   284→      <SaleFormDialog
   285→        open={formOpen}
   286→        onOpenChange={setFormOpen}
   287→        onCreated={(sale) => { setInvoiceSale(sale); qc.invalidateQueries({ queryKey: ["sales"] }); qc.invalidateQueries({ queryKey: ["products"] }); }}
   288→      />
   289→
   290→      <InvoiceDialog sale={invoiceSale} onOpenChange={(o) => !o && setInvoiceSale(null)} />
   291→    </div>
   292→  );
   293→}

   295→// ────────────────────────────────────────────────────────────────────────────
   296→// Sale Form Dialog (POS-style)
   297→// ────────────────────────────────────────────────────────────────────────────
   298→function SaleFormDialog({
   299→  open, onOpenChange, onCreated,
   300→}: {
   301→  open: boolean;
   302→  onOpenChange: (o: boolean) => void;
   303→  onCreated: (sale: Sale) => void;
   304→}) {
   305→  const qc = useQueryClient();
   306→  const [customerId, setCustomerId] = useState("");
   307→  const [cart, setCart] = useState<CartLine[]>([]);
   308→  const [search, setSearch] = useState("");
   309→  const [overallDiscount, setOverallDiscount] = useState(0);
   310→  const [tax, setTax] = useState(0);
   311→  const [paymentMethod, setPaymentMethod] = useState("CASH");
   312→  const [paymentStatus, setPaymentStatus] = useState("PAID");
   313→  const [notes, setNotes] = useState("");
   314→
   315→  const customers = useQuery<Customer[]>({
   316→    queryKey: ["customers"],
   317→    queryFn: () => api.get("/customers"),
   318→    staleTime: 60_000,
   319→  });
   320→
   321→  const productSearch = useQuery<{ data: Product[] }>({
   322→    queryKey: ["products-pos", search],
   323→    queryFn: () => api.get(`/products?q=${encodeURIComponent(search)}&pageSize=10`),
   324→    enabled: open && search.length > 0,
   325→    staleTime: 10_000,
   326→  });
   327→
   328→  const resetForm = () => {
   329→    setCustomerId(""); setCart([]); setSearch(""); setOverallDiscount(0);
   330→    setTax(0); setPaymentMethod("CASH"); setPaymentStatus("PAID"); setNotes("");
   331→  };
   332→  const handleOpenChange = (o: boolean) => {
   333→    if (!o) resetForm();
   334→    onOpenChange(o);
   335→  };
   336→
   337→  const totals = useMemo(() => {
   338→    const subtotal = cart.reduce((s, l) => s + l.qty * l.price, 0);
   339→    const lineDiscounts = cart.reduce((s, l) => s + l.discount, 0);
   340→    const cost = cart.reduce((s, l) => s + l.qty * l.cost, 0);
   341→    const net = subtotal - lineDiscounts - overallDiscount;
   342→    const total = Math.max(0, net + tax);
   343→    const profit = net - cost;
   344→    return { subtotal, lineDiscounts, cost, net, total, profit };
   345→  }, [cart, overallDiscount, tax]);
   346→
   347→  const addToCart = (p: Product) => {
   348→    setCart((prev) => {
   349→      const existing = prev.find((l) => l.productId === p.id);
   350→      if (existing) {
   351→        return prev.map((l) => l.productId === p.id ? { ...l, qty: l.qty + 1 } : l);
   352→      }
   353→      return [...prev, {
   354→        key: p.id, productId: p.id, name: p.name, sku: p.sku,
   355→        qty: 1, price: p.sellingPrice, cost: p.purchasePrice, discount: 0, stock: p.stock,
   356→      }];
   357→    });
   358→    setSearch("");
   359→  };
   360→
   361→  // ── Scan handler — search by detected code & auto-add if exactly one match ─
   362→  const handleScanDetected = async (code: string) => {
   363→    const trimmed = code.trim();
   364→    if (!trimmed) return;
   365→    try {
   366→      const res = await api.get<{ data: Product[] }>(
   367→        `/products?q=${encodeURIComponent(trimmed)}&pageSize=10`,
   368→      );
   369→      const matches = res?.data ?? [];
   370→      if (matches.length === 1) {
   371→        addToCart(matches[0]);
   372→        toast.success(`Added "${matches[0].name}" to cart`);
   373→      } else if (matches.length > 1) {
   374→        setSearch(trimmed);
   375→        toast.info(`${matches.length} products match code "${trimmed}" — pick one below.`);
   376→      } else {
   377→        toast.error(`No product found for code "${trimmed}"`);
   378→      }
   379→    } catch (e) {
   380→      toast.error((e as Error).message);
   381→    }
   382→  };
   383→
   384→  const updateLine = (key: string, patch: Partial<CartLine>) => {
   385→    setCart((prev) => prev.map((l) => l.key === key ? { ...l, ...patch } : l));
   386→  };
   387→  const removeLine = (key: string) => setCart((prev) => prev.filter((l) => l.key !== key));
   388→
   389→  const createSale = useMutation({
   390→    mutationFn: () => api.post<Sale>("/sales", {
   391→      customerId: customerId || undefined,
   392→      items: cart.map((l) => ({ productId: l.productId, qty: l.qty, price: l.price, discount: l.discount })),
   393→      discount: overallDiscount,
   394→      tax,
   395→      paymentMethod,
   396→      paymentStatus,
   397→      notes,
   398→    }),
   399→    onSuccess: (sale) => {
   400→      toast.success(`Sale ${sale.invoiceNo} created`);
   401→      qc.invalidateQueries({ queryKey: ["sales"] });
   402→      qc.invalidateQueries({ queryKey: ["products"] });
   403→      onOpenChange(false);
   404→      onCreated(sale);
   405→    },
   406→    onError: (e: Error) => toast.error(e.message),
   407→  });
   408→
   409→  const canSubmit = cart.length > 0 && !createSale.isPending;
   410→
   411→  return (
   412→    <Dialog open={open} onOpenChange={handleOpenChange}>
   413→      <DialogContent className="sm:max-w-4xl max-h-[92vh] overflow-hidden p-0 gap-0">
   414→        <DialogHeader className="px-6 py-4 border-b bg-muted/30">
   415→          <DialogTitle className="flex items-center gap-2">
   416→            <ShoppingCart className="h-5 w-5 text-primary" /> New Sale
   417→          </DialogTitle>
   418→          <DialogDescription>Add products to cart and checkout. Stock is deducted automatically.</DialogDescription>
   419→        </DialogHeader>
   420→
   421→        <div className="grid grid-cols-1 lg:grid-cols-[1fr_360px] overflow-hidden">
   422→          {/* Left: product search + cart */}
   423→          <div className="flex flex-col overflow-hidden border-r">
   424→            {/* Search */}
   425→            <div className="p-4 border-b">
   426→              <div className="flex gap-2">
   427→                <div className="relative flex-1">
   428→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   429→                  <Input
   430→                    value={search}
   431→                    onChange={(e) => setSearch(e.target.value)}
   432→                    placeholder="Search products by name, SKU, barcode…"
   433→                    className="pl-9"
   434→                    autoFocus
   435→                  />
   436→                </div>
   437→                <ScannerButton
   438→                  label="Scan"
   439→                  onDetected={handleScanDetected}
   440→                  className="shrink-0"
   441→                />
   442→              </div>
   443→              {/* Search results */}
   444→              {search && (
   445→                <div className="mt-2 max-h-64 overflow-y-auto rounded-lg border bg-popover shadow-sm">
   446→                  {productSearch.isLoading && (
   447→                    <div className="p-3 text-xs text-muted-foreground flex items-center gap-2">
   448→                      <Loader2 className="h-3 w-3 animate-spin" /> Searching…
   449→                    </div>
   450→                  )}
   451→                  {!productSearch.isLoading && (productSearch.data?.data ?? []).length === 0 && (
   452→                    <div className="p-3 text-xs text-muted-foreground">No products found.</div>
   453→                  )}
   454→                  <AnimatePresence>
   455→                    {(productSearch.data?.data ?? []).map((p) => (
   456→                      <motion.button
   457→                        key={p.id}
   458→                        initial={{ opacity: 0, y: -4 }}
   459→                        animate={{ opacity: 1, y: 0 }}
   460→                        exit={{ opacity: 0 }}
   461→                        onClick={() => addToCart(p)}
   462→                        className="flex w-full items-center gap-3 border-b px-3 py-2 text-left last:border-b-0 hover:bg-accent transition"
   463→                      >
   464→                        <div className="flex h-9 w-9 items-center justify-center rounded-md bg-primary/10 text-primary">
   465→                          <Package className="h-4 w-4" />
   466→                        </div>
   467→                        <div className="min-w-0 flex-1">
   468→                          <p className="truncate text-sm font-medium">{p.name}</p>
   469→                          <p className="text-[11px] text-muted-foreground">
   470→                            {p.sku} · {p.brand?.name ?? ""} {p.model?.name ?? ""}
   471→                          </p>
   472→                        </div>
   473→                        <div className="text-right">
   474→                          <p className="text-sm font-semibold text-emerald-600">{formatCurrency(p.sellingPrice)}</p>
   475→                          <p className="text-[11px] text-muted-foreground">{p.stock} in stock</p>
   476→                        </div>
   477→                        <Plus className="h-4 w-4 text-muted-foreground" />
   478→                      </motion.button>
   479→                    ))}
   480→                  </AnimatePresence>
   481→                </div>
   482→              )}
   483→            </div>
   484→
   485→            {/* Cart list */}
   486→            <div className="flex-1 overflow-hidden">
   487→              {cart.length === 0 ? (

   893→                  </tbody>
   894→                </table>
   895→              </div>
   896→
   897→              {/* Totals */}
   898→              <div className="ml-auto w-full max-w-xs space-y-1.5 text-sm">
   899→                <Row label="Subtotal" value={formatCurrency(sale.subtotal, sale.business?.currencySymbol ?? "Rs")} />
   900→                {sale.discount > 0 && <Row label="Discount" value={`− ${formatCurrency(sale.discount, sale.business?.currencySymbol ?? "Rs")}`} muted />}
   901→                {sale.tax > 0 && <Row label="Tax" value={`+ ${formatCurrency(sale.tax, sale.business?.currencySymbol ?? "Rs")}`} />}
   902→                <div className="my-1 border-t" />
   903→                <div className="flex items-center justify-between">
   904→                  <span className="font-semibold">Total</span>
   905→                  <span className="text-lg font-bold text-primary">{formatCurrency(sale.total, sale.business?.currencySymbol ?? "Rs")}</span>
   906→                </div>
   907→                {sale.paid > 0 && <Row label="Paid" value={formatCurrency(sale.paid, sale.business?.currencySymbol ?? "Rs")} />}
   908→                {sale.total - sale.paid > 0 && (
   909→                  <div className="flex items-center justify-between rounded-md bg-amber-500/10 px-2 py-1.5 text-amber-700 dark:text-amber-400">
   910→                    <span className="text-xs font-medium">Balance Due</span>
   911→                    <span className="font-semibold">{formatCurrency(sale.total - sale.paid, sale.business?.currencySymbol ?? "Rs")}</span>
   912→                  </div>
   913→                )}
   914→                <div className="flex items-center justify-between pt-1 text-xs">
   915→                  <span className="text-muted-foreground">Est. profit</span>
   916→                  <span className={`font-semibold ${sale.profit >= 0 ? "text-emerald-600" : "text-rose-600"}`}>
   917→                    {formatCurrency(sale.profit, sale.business?.currencySymbol ?? "Rs")}
   918→                  </span>
   919→                </div>
   920→              </div>
   921→
   922→              {sale.notes && (
   923→                <div className="rounded-lg border border-amber-500/20 bg-amber-500/5 p-3">
   924→                  <p className="mb-1 text-[11px] font-semibold uppercase text-amber-600">Notes</p>
   925→                  <p className="text-sm">{sale.notes}</p>
   926→                </div>
   927→              )}
   928→
   929→              {sale.user && (
   930→                <p className="text-center text-[11px] text-muted-foreground">Served by {sale.user.name}</p>
   931→              )}
   932→            </div>
   933→          </ScrollArea>
   934→        )}
   935→      </DialogContent>
   936→    </Dialog>
   937→  );
   938→}
