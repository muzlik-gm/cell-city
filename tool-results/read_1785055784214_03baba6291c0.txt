   154→// Visual status timeline stepper — RECEIVED → DIAGNOSED → … → DELIVERED
   155→function StatusTimeline({ status }: { status: string }) {
   156→  const currentIdx = STATUS_FLOW.indexOf(status as RepairStatus);
   157→  const isCancelled = status === "CANCELLED";
   158→  if (isCancelled) {
   159→    return (
   160→      <div className="flex items-center gap-2 rounded-lg border border-rose-500/30 bg-rose-500/5 px-3 py-2.5">
   161→        <Ban className="h-4 w-4 text-rose-500" />
   162→        <p className="text-sm font-medium text-rose-600 dark:text-rose-400">This ticket was cancelled</p>
   163→      </div>
   164→    );
   165→  }
   166→  return (
   167→    <div className="flex items-center justify-between gap-1">
   168→      {STATUS_FLOW.map((s, i) => {
   169→        const meta = COLUMN_META[s];
   170→        const isDone = i < currentIdx;
   171→        const isCurrent = i === currentIdx;
   172→        const isFuture = i > currentIdx;
   173→        const Icon = meta.icon;
   174→        return (
   175→          <div key={s} className="flex flex-1 flex-col items-center">
   176→            <div className="flex w-full items-center">
   177→              {/* connector before */}
   178→              <div className={`h-0.5 flex-1 ${i === 0 ? "opacity-0" : isDone || isCurrent ? meta.dot : "bg-border"}`} />
   179→              {/* circle */}
   180→              <div
   181→                className={`relative flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition ${
   182→                  isCurrent
   183→                    ? `${meta.dot} border-transparent text-white shadow-md`
   184→                    : isDone
   185→                    ? `${meta.dot} border-transparent text-white`
   186→                    : "border-border bg-background text-muted-foreground"
   187→                }`}
   188→              >
   189→                {isDone ? <CheckCheck className="h-4 w-4" /> : <Icon className="h-3.5 w-3.5" />}
   190→                {isCurrent && (
   191→                  <span className={`absolute -inset-1 animate-ping rounded-full ${meta.dot} opacity-30`} />
   192→                )}
   193→              </div>
   194→              {/* connector after */}
   195→              <div className={`h-0.5 flex-1 ${i === STATUS_FLOW.length - 1 ? "opacity-0" : isDone ? meta.dot : "bg-border"}`} />
   196→            </div>
   197→            <p
   198→              className={`mt-1.5 text-center text-[10px] font-medium leading-tight ${
   199→                isCurrent ? meta.accent : isFuture ? "text-muted-foreground/60" : "text-muted-foreground"
   200→              }`}
   201→            >
   202→              {meta.label}
   203→            </p>
   204→          </div>
   205→        );
   206→      })}
   207→    </div>
   208→  );
   209→}

   220→function NewTicketDialog({
   221→  open, onOpenChange, customers, models, technicians,
   222→}: {
   223→  open: boolean;
   224→  onOpenChange: (v: boolean) => void;
   225→  customers: Customer[];
   226→  models: PhoneModel[];
   227→  technicians: User[];
   228→}) {
   229→  const qc = useQueryClient();
   230→  const [customerId, setCustomerId] = useState("");
   231→  const [modelId, setModelId] = useState("");
   232→  const [technicianId, setTechnicianId] = useState("");
   233→  const [imei, setImei] = useState("");
   234→  const [problem, setProblem] = useState("");
   235→  const [diagnosis, setDiagnosis] = useState("");
   236→  const [laborCost, setLaborCost] = useState("");
   237→  const [notes, setNotes] = useState("");
   238→  const [imageUrl, setImageUrl] = useState<string | null>(null);
   239→  const [saving, setSaving] = useState(false);
   240→
   241→  const reset = useCallback(() => {
   242→    setCustomerId(""); setModelId(""); setTechnicianId(""); setImei("");
   243→    setProblem(""); setDiagnosis(""); setLaborCost(""); setNotes(""); setImageUrl(null);
   244→  }, []);
   245→
   246→  const handleSubmit = async () => {
   247→    if (!problem.trim()) { toast.error("Problem description is required"); return; }
   248→    setSaving(true);
   249→    try {
   250→      const payload: NewTicketPayload = {
   251→        problem: problem.trim(),
   252→        ...(customerId ? { customerId } : {}),
   253→        ...(modelId ? { modelId } : {}),
   254→        ...(technicianId ? { technicianId } : {}),
   255→        ...(imei.trim() ? { imei: imei.trim() } : {}),
   256→        ...(diagnosis.trim() ? { diagnosis: diagnosis.trim() } : {}),
   257→        ...(laborCost ? { laborCost: Number(laborCost) } : {}),
   258→        ...(notes.trim() ? { notes: notes.trim() } : {}),
   259→        ...(imageUrl ? { imageUrl } : {}),
   260→      };

   262→      toast.success(`Ticket ${created.ticketNo} created`);
   263→      qc.invalidateQueries({ queryKey: ["repairs"] });
   264→      qc.invalidateQueries({ queryKey: ["repairs-stats"] });
   265→      reset();
   266→      onOpenChange(false);
   267→    } catch (e) {
   268→      toast.error((e as Error).message);

   270→      setSaving(false);
   271→    }
   272→  };

   285→        </DialogHeader>
   286→
   287→        <ScrollArea className="max-h-[70vh]">
   288→          <div className="space-y-4 p-5">
   289→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   290→              <div className="space-y-1.5">
   291→                <Label className="text-xs font-medium">Customer</Label>
   292→                <Select value={customerId} onValueChange={setCustomerId}>
   293→                  <SelectTrigger><SelectValue placeholder="Walk-in customer" /></SelectTrigger>
   294→                  <SelectContent>
   295→                    {customers.map((c) => (
   296→                      <SelectItem key={c.id} value={c.id}>{c.name}{c.phone ? ` · ${c.phone}` : ""}</SelectItem>
   297→                    ))}
   298→                  </SelectContent>
   299→                </Select>
   300→              </div>
   301→              <div className="space-y-1.5">
   302→                <Label className="text-xs font-medium">Phone Model</Label>
   303→                <Select value={modelId} onValueChange={setModelId}>

   310→                    ))}
   311→                  </SelectContent>
   312→                </Select>
   313→              </div>
   314→              <div className="space-y-1.5">
   315→                <Label className="text-xs font-medium">Technician</Label>

   395→function AddPartDialog({
   396→  open, onOpenChange, repairId,
   397→}: { open: boolean; onOpenChange: (v: boolean) => void; repairId: string }) {
   398→  const qc = useQueryClient();
   399→  const [search, setSearch] = useState("");
   400→  const [selectedId, setSelectedId] = useState("");
   401→  const [qty, setQty] = useState("1");
   402→  const [used, setUsed] = useState(true);
   403→  const [saving, setSaving] = useState(false);
   404→
   405→  const productsQ = useQuery<{ data: Product[] }>({
   406→    queryKey: ["repair-parts-search", search],
   407→    queryFn: () => api.get(`/products?q=${encodeURIComponent(search)}&pageSize=30`),
   408→    enabled: open,
   409→    staleTime: 10_000,
   410→  });
   411→  const products = productsQ.data?.data ?? [];
   412→
   413→  const handleSubmit = async () => {
   414→    if (!selectedId) { toast.error("Select a product"); return; }
   415→    const q = Math.max(1, parseInt(qty, 10) || 1);
   416→    setSaving(true);
   417→    try {
   418→      await api.post(`/repairs/${repairId}/parts`, { productId: selectedId, qty: q, used });
   419→      toast.success("Part added to repair");
   420→      qc.invalidateQueries({ queryKey: ["repair-detail", repairId] });
   421→      qc.invalidateQueries({ queryKey: ["repairs"] });
   422→      setSearch(""); setSelectedId(""); setQty("1"); setUsed(true);
   423→      onOpenChange(false);
   424→    } catch (e) {
   425→      toast.error((e as Error).message);
   426→    } finally {
   427→      setSaving(false);
   428→    }
   429→  };
   430→
   431→  return (
   432→    <Dialog open={open} onOpenChange={onOpenChange}>
   433→      <DialogContent className="max-w-lg p-0 gap-0 overflow-hidden">
   434→        <DialogHeader className="border-b bg-muted/30 p-5">
   435→          <DialogTitle className="flex items-center gap-2">
   436→            <Package className="h-4 w-4 text-primary" /> Add Part from Inventory
   437→          </DialogTitle>
   438→          <DialogDescription>Select a part to attach to this repair ticket.</DialogDescription>
   439→        </DialogHeader>
   440→        <div className="space-y-4 p-5">
   441→          <div className="relative">
   442→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   443→            <Input
   444→              value={search} onChange={(e) => setSearch(e.target.value)}
   445→              placeholder="Search products by name, SKU, model..."
   446→              className="pl-9"
   447→            />
   448→          </div>
   449→          <div className="max-h-64 space-y-1 overflow-y-auto rounded-lg border">
   450→            {productsQ.isLoading ? (

   487→                  <SelectItem value="used">Deduct now</SelectItem>
   488→                  <SelectItem value="reserved">Reserve only</SelectItem>

   490→              </Select>
   491→            </div>
   492→          </div>
   493→        </div>

   499→          </Button>

   502→    </Dialog>

   509→function RepairDetailSheet({

   512→  const qc = useQueryClient();

   514→  const [editDiagnosis, setEditDiagnosis] = useState(false);
   515→  const [diagnosisDraft, setDiagnosisDraft] = useState("");

   526→    queryFn: () => api.get(`/repairs/${repairId}`),
   527→    enabled: open,
   528→  });
   529→  const r = detailQ.data;
   530→
   531→  const updateMutation = useMutation({
   532→    mutationFn: async (body: Record<string, unknown>) => {
   533→      if (!repairId) throw new Error("No repair selected");
   534→      return api.patch<RepairJob>(`/repairs/${repairId}`, body);
   535→    },
   536→    onSuccess: () => {
   537→      qc.invalidateQueries({ queryKey: ["repair-detail", repairId] });
   538→      qc.invalidateQueries({ queryKey: ["repairs"] });
   539→      qc.invalidateQueries({ queryKey: ["repairs-stats"] });

   541→    onError: (e: Error) => toast.error(e.message),
   542→  });
   543→
   544→  const togglePartUsed = async (part: RepairPart) => {
   545→    try {
   546→      await api.patch(`/repairs/${repairId}/parts?partId=${part.id}`, { used: !part.used });
   547→      toast.success(part.used ? "Stock restocked" : "Marked as used · stock deducted");
   548→      qc.invalidateQueries({ queryKey: ["repair-detail", repairId] });
   549→      qc.invalidateQueries({ queryKey: ["repairs"] });
   550→    } catch (e) {
   551→      toast.error((e as Error).message);
   552→    }
   553→  };
   554→
   555→  const removePart = async (part: RepairPart) => {
   556→    if (!confirm(`Remove ${part.product?.name ?? "part"} from this repair?${part.used ? " Stock will be restocked." : ""}`)) return;
   557→    try {
   558→      await api.del(`/repairs/${repairId}/parts?partId=${part.id}`);
   559→      toast.success("Part removed");
   560→      qc.invalidateQueries({ queryKey: ["repair-detail", repairId] });
   561→      qc.invalidateQueries({ queryKey: ["repairs"] });
   562→    } catch (e) {
   563→      toast.error((e as Error).message);
   564→    }
   565→  };
   566→
   567→  const handleDelete = async () => {
   568→    if (!r) return;
   569→    if (!confirm(`Delete ticket ${r.ticketNo}? This cannot be undone.`)) return;
   570→    try {
   571→      await api.del(`/repairs/${r.id}`);
   572→      toast.success("Repair ticket deleted");
   573→      qc.invalidateQueries({ queryKey: ["repairs"] });
   574→      qc.invalidateQueries({ queryKey: ["repairs-stats"] });
   575→      onClose();

   577→      toast.error((e as Error).message);
   578→    }
   579→  };
   580→
   581→  const saveDiagnosis = () => {
   582→    updateMutation.mutate({ diagnosis: diagnosisDraft }, {
   583→      onSuccess: () => { toast.success("Diagnosis updated"); setEditDiagnosis(false); },
   584→    });

   586→  const saveCosts = () => {

   588→      onSuccess: () => { toast.success("Labor cost updated"); setEditCosts(false); },

   590→  };

   595→    );
   596→  };
   597→
   598→  return (
   599→    <Sheet open={open} onOpenChange={(v) => { if (!v) onClose(); }}>
   600→      <SheetContent side="right" className="w-full gap-0 p-0 sm:max-w-xl md:max-w-2xl">
   601→        <SheetHeader className="border-b bg-muted/30 p-5">
   602→          {r ? (
   603→            <>
   604→              <div className="flex items-center justify-between pr-6">
   605→                <div className="min-w-0">
   606→                  <SheetTitle className="flex items-center gap-2 font-mono text-base">
   607→                    {r.ticketNo}
   608→                  </SheetTitle>
   609→                  <SheetDescription className="mt-0.5">
   610→                    Opened {timeAgo(r.receivedAt)} · {formatDateTime(r.receivedAt)}
   611→                  </SheetDescription>
   612→                </div>
   613→                <RepairStatusBadge status={r.status} />
   614→              </div>
   615→            </>
   616→          ) : (
   617→            <div className="space-y-2">
   618→              <div className="h-5 w-32 animate-pulse rounded bg-muted" />
   619→              <div className="h-3 w-48 animate-pulse rounded bg-muted" />
   620→            </div>
   621→          )}
   622→        </SheetHeader>
   623→
   624→        {detailQ.isLoading ? (
   625→          <LoadingState className="flex-1" />
   626→        ) : detailQ.isError ? (
   627→          <div className="flex-1 p-6 text-sm text-rose-600">Failed to load repair.</div>
   628→        ) : !r ? null : (
   629→          <ScrollArea className="flex-1">
   630→            <div className="space-y-5 p-5">
   631→              {/* Status timeline */}
   632→              <div>
   633→                <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Status Timeline</p>
   634→                <StatusTimeline status={r.status} />
   635→              </div>
   636→
   637→              {/* Quick status changer */}
   638→              <div className="flex flex-wrap items-center gap-2 rounded-xl border bg-card p-3">
   639→                <span className="text-xs font-medium text-muted-foreground">Move to:</span>
   640→                <Select
   641→                  value={r.status}
   642→                  onValueChange={(v) => updateMutation.mutate({ status: v })}
   643→                >
   644→                  <SelectTrigger className="h-8 w-44 text-xs">
   645→                    <SelectValue />
   646→                  </SelectTrigger>
   647→                  <SelectContent>
