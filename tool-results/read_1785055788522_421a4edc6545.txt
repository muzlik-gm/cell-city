   154→   154→// Visual status timeline stepper — RECEIVED → DIAGNOSED → … → DELIVERED
   155→   155→function StatusTimeline({ status }: { status: string }) {
   156→   156→  const currentIdx = STATUS_FLOW.indexOf(status as RepairStatus);
   157→   157→  const isCancelled = status === "CANCELLED";
   158→   158→  if (isCancelled) {
   159→   159→    return (
   160→   160→      <div className="flex items-center gap-2 rounded-lg border border-rose-500/30 bg-rose-500/5 px-3 py-2.5">
   161→   161→        <Ban className="h-4 w-4 text-rose-500" />
   162→   162→        <p className="text-sm font-medium text-rose-600 dark:text-rose-400">This ticket was cancelled</p>
   163→   163→      </div>
   164→   164→    );
   165→   165→  }
   166→   166→  return (
   167→   167→    <div className="flex items-center justify-between gap-1">
   168→   168→      {STATUS_FLOW.map((s, i) => {
   169→   169→        const meta = COLUMN_META[s];
   170→   170→        const isDone = i < currentIdx;
   171→   171→        const isCurrent = i === currentIdx;
   172→   172→        const isFuture = i > currentIdx;
   173→   173→        const Icon = meta.icon;
   174→   174→        return (
   175→   175→          <div key={s} className="flex flex-1 flex-col items-center">
   176→   176→            <div className="flex w-full items-center">
   177→   177→              {/* connector before */}
   178→   178→              <div className={`h-0.5 flex-1 ${i === 0 ? "opacity-0" : isDone || isCurrent ? meta.dot : "bg-border"}`} />
   179→   179→              {/* circle */}
   180→   180→              <div
   181→   181→                className={`relative flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition ${
   182→   182→                  isCurrent
   183→   183→                    ? `${meta.dot} border-transparent text-white shadow-md`
   184→   184→                    : isDone
   185→   185→                    ? `${meta.dot} border-transparent text-white`
   186→   186→                    : "border-border bg-background text-muted-foreground"
   187→   187→                }`}
   188→   188→              >
   189→   189→                {isDone ? <CheckCheck className="h-4 w-4" /> : <Icon className="h-3.5 w-3.5" />}
   190→   190→                {isCurrent && (
   191→   191→                  <span className={`absolute -inset-1 animate-ping rounded-full ${meta.dot} opacity-30`} />
   192→   192→                )}
   193→   193→              </div>
   194→   194→              {/* connector after */}
   195→   195→              <div className={`h-0.5 flex-1 ${i === STATUS_FLOW.length - 1 ? "opacity-0" : isDone ? meta.dot : "bg-border"}`} />
   196→   196→            </div>
   197→   197→            <p
   198→   198→              className={`mt-1.5 text-center text-[10px] font-medium leading-tight ${
   199→   199→                isCurrent ? meta.accent : isFuture ? "text-muted-foreground/60" : "text-muted-foreground"
   200→   200→              }`}
   201→   201→            >
   202→   202→              {meta.label}
   203→   203→            </p>
   204→   204→          </div>
   205→   205→        );
   206→   206→      })}
   207→   207→    </div>
   208→   208→  );
   209→   209→}

   214→   214→interface NewTicketPayload {

   220→   220→function NewTicketDialog({
   221→   221→  open, onOpenChange, customers, models, technicians,
   222→   222→}: {
   223→   223→  open: boolean;
   224→   224→  onOpenChange: (v: boolean) => void;
   225→   225→  customers: Customer[];
   226→   226→  models: PhoneModel[];
   227→   227→  technicians: User[];
   228→   228→}) {
   229→   229→  const qc = useQueryClient();
   230→   230→  const [customerId, setCustomerId] = useState("");
   231→   231→  const [modelId, setModelId] = useState("");
   232→   232→  const [technicianId, setTechnicianId] = useState("");
   233→   233→  const [imei, setImei] = useState("");
   234→   234→  const [problem, setProblem] = useState("");
   235→   235→  const [diagnosis, setDiagnosis] = useState("");
   236→   236→  const [laborCost, setLaborCost] = useState("");
   237→   237→  const [notes, setNotes] = useState("");
   238→   238→  const [imageUrl, setImageUrl] = useState<string | null>(null);
   239→   239→  const [saving, setSaving] = useState(false);
   240→   240→
   241→   241→  const reset = useCallback(() => {
   242→   242→    setCustomerId(""); setModelId(""); setTechnicianId(""); setImei("");
   243→   243→    setProblem(""); setDiagnosis(""); setLaborCost(""); setNotes(""); setImageUrl(null);
   244→   244→  }, []);
   245→   245→
   246→   246→  const handleSubmit = async () => {
   247→   247→    if (!problem.trim()) { toast.error("Problem description is required"); return; }
   248→   248→    setSaving(true);
   249→   249→    try {
   250→   250→      const payload: NewTicketPayload = {
   251→   251→        problem: problem.trim(),
   252→   252→        ...(customerId ? { customerId } : {}),
   253→   253→        ...(modelId ? { modelId } : {}),
   254→   254→        ...(technicianId ? { technicianId } : {}),
   255→   255→        ...(imei.trim() ? { imei: imei.trim() } : {}),
   256→   256→        ...(diagnosis.trim() ? { diagnosis: diagnosis.trim() } : {}),
   257→   257→        ...(laborCost ? { laborCost: Number(laborCost) } : {}),
   258→   258→        ...(notes.trim() ? { notes: notes.trim() } : {}),
   259→   259→        ...(imageUrl ? { imageUrl } : {}),
   260→   260→      };
   261→   261→      const created = await api.post<RepairJob>("/repairs", payload);
   262→   262→      toast.success(`Ticket ${created.ticketNo} created`);
   263→   263→      qc.invalidateQueries({ queryKey: ["repairs"] });
   264→   264→      qc.invalidateQueries({ queryKey: ["repairs-stats"] });
   265→   265→      reset();
   266→   266→      onOpenChange(false);
   267→   267→    } catch (e) {
   268→   268→      toast.error((e as Error).message);
   269→   269→    } finally {
   270→   270→      setSaving(false);
   271→   271→    }
   272→   272→  };
   273→   273→
   274→   274→  return (
   275→   275→    <Dialog open={open} onOpenChange={(v) => { if (!v) reset(); onOpenChange(v); }}>
   276→   276→      <DialogContent className="max-w-2xl gap-0 overflow-hidden p-0 sm:max-w-2xl">
   277→   277→        <DialogHeader className="border-b bg-muted/30 p-5">
   278→   278→          <DialogTitle className="flex items-center gap-2">
   279→   279→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
   280→   280→              <Wrench className="h-4 w-4" />
   281→   281→            </div>
   282→   282→            New Repair Ticket
   283→   283→          </DialogTitle>
   284→   284→          <DialogDescription>Create a new repair job. Ticket number is auto-generated.</DialogDescription>
   285→   285→        </DialogHeader>
   286→   286→
   287→   287→        <ScrollArea className="max-h-[70vh]">
   288→   288→          <div className="space-y-4 p-5">
   289→   289→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   290→   290→              <div className="space-y-1.5">
   291→   291→                <Label className="text-xs font-medium">Customer</Label>
   292→   292→                <Select value={customerId} onValueChange={setCustomerId}>
   293→   293→                  <SelectTrigger><SelectValue placeholder="Walk-in customer" /></SelectTrigger>
   294→   294→                  <SelectContent>
   295→   295→                    {customers.map((c) => (
   296→   296→                      <SelectItem key={c.id} value={c.id}>{c.name}{c.phone ? ` · ${c.phone}` : ""}</SelectItem>
   297→   297→                    ))}
   298→   298→                  </SelectContent>
   299→   299→                </Select>
   300→   300→              </div>
   301→   301→              <div className="space-y-1.5">
   302→   302→                <Label className="text-xs font-medium">Phone Model</Label>
   303→   303→                <Select value={modelId} onValueChange={setModelId}>
   304→   304→                  <SelectTrigger><SelectValue placeholder="Select model" /></SelectTrigger>
   305→   305→                  <SelectContent className="max-h-64">
   306→   306→                    {models.map((m) => (
   307→   307→                      <SelectItem key={m.id} value={m.id}>
   308→   308→                        {m.brand?.name ? `${m.brand.name} ` : ""}{m.name}
   309→   309→                      </SelectItem>
   310→   310→                    ))}
   311→   311→                  </SelectContent>
